/********************* PHASE 1: PERF INSTRUMENTATION ***********
 * Đo lường nhẹ + chẩn đoán kích thước cache. Không ghi dữ liệu,
 * không log token/mật khẩu/payload đầy đủ. Chỉ log khi có lỗi cache
 * hoặc request chậm vượt ngưỡng, kèm mẫu nhỏ để tránh spam log.
 * Rollback: xóa nguyên khối này + action 'getPerfDiagnostics' trong api().
 **************************************************************/
const PERF_SLOW_MS_ = 3000;               // Ngưỡng coi là "chậm" để ghi log.
const PERF_SAMPLE_RATE_ = 0.02;            // Lấy mẫu nhỏ request bình thường để vẫn có điểm đo bytes.
const PERF_CACHE_VALUE_LIMIT_ = 100 * 1024; // Giới hạn ~100KB/value của CacheService.
const PERF_CHUNK_MAX_BYTES_ = 80 * 1024;    // Trần an toàn mỗi chunk (UTF-8).

// Đo UTF-8 bytes chính xác trong môi trường Apps Script.
function utf8ByteLen_(s) {
  try { return Utilities.newBlob(String(s == null ? '' : s)).getBytes().length; }
  catch (e) { return String(s == null ? '' : s).length; }
}

// Chuỗi lỗi rút gọn, an toàn — không bao giờ chứa payload đầy đủ.
function perfSafeErr_(e) {
  try {
    const msg = (e && e.message) ? e.message : String(e);
    return safeStr_(msg).slice(0, 180);
  } catch (ignore) { return 'unknown_error'; }
}

// Bộ đếm thời gian nhẹ theo mốc. mark() ghi mốc, snapshot() trả tổng hợp ms.
function perfTimer_() {
  const t0 = Date.now();
  const marks = {};
  let last = t0;
  return {
    mark: function (name) {
      const now = Date.now();
      marks[name] = (marks[name] || 0) + (now - last);
      last = now;
    },
    totalMs: function () { return Date.now() - t0; },
    snapshot: function () {
      const out = {}; Object.keys(marks).forEach(function (k) { out[k] = marks[k]; });
      out.totalMs = Date.now() - t0;
      return out;
    }
  };
}

// Lấy mẫu nhỏ request bình thường để vẫn có điểm đo bytes mà không phải stringify mọi lượt gọi.
function perfShouldSample_() {
  try {
    const rate = Number(PERF_SAMPLE_RATE_) || 0;
    if (!(rate > 0)) return false;
    const raw = safeStr_(Utilities.getUuid()).replace(/-/g, '');
    const bucket = parseInt(raw.slice(0, 2) || 'ff', 16);
    return bucket < Math.max(1, Math.round(rate * 256));
  } catch (e) { return false; }
}

function perfShouldLog_(timing, errStr, sampled) {
  const total = timing && timing.totalMs ? timing.totalMs : 0;
  return !!errStr || total >= PERF_SLOW_MS_ || !!sampled;
}

// Chỉ log khi lỗi/chậm/lấy mẫu; không log dữ liệu nhạy cảm hay payload đầy đủ.
function perfLog_(action, cacheState, sizeBytes, timing, errStr, sampled) {
  try {
    const total = timing && timing.totalMs ? timing.totalMs : 0;
    const isSlow = total >= PERF_SLOW_MS_;
    if (!errStr && !isSlow && !sampled) return; // bỏ qua request bình thường để tránh spam.
    const parts = [
      'PERF', safeStr_(action) || '(unknown)',
      'cache=' + safeStr_(cacheState || 'n/a'),
      'bytes=' + (Number(sizeBytes) || 0),
      'ms=' + total
    ];
    if (timing) {
      const keys = Object.keys(timing).filter(function (k) { return k !== 'totalMs'; });
      if (keys.length) parts.push('t={' + keys.map(function (k) { return k + ':' + timing[k]; }).join(',') + '}');
    }
    if (sampled) parts.push('sampled=1');
    if (errStr) parts.push('err=' + safeStr_(errStr).slice(0, 180));
    Logger.log(parts.join(' '));
  } catch (ignore) {}
}

// Chẩn đoán kích thước payload danh sách lịch: đếm bản ghi thật, byte UTF-8,
// vượt 100KB hay không, và các trường chiếm dung lượng lớn nhất.
// KHÔNG quét Drive: dựng meetingRow_ với fileCountMap=null để đo đúng phần dữ liệu sheet.
function getPerfDiagnostics_(p) {
  const user = requireAuth_(p);
  if (!isAdminLike_(user)) return { ok: false, msg: 'Chỉ quản trị mới xem được chẩn đoán hiệu năng.' };

  const timer = perfTimer_();
  let meetings = [];
  try {
    const t = tableFast_(CFG.SH_MEETING, ['MEETING_ID'], [], true);
    timer.mark('readMeetingSheet');
    meetings = t.rows.map(function (r) { return meetingRow_(r, null); }).filter(function (m) { return !m.isDeleted; });
    timer.mark('buildRows');
  } catch (e) {
    perfLog_('getPerfDiagnostics', 'ERR', 0, timer.snapshot(), perfSafeErr_(e));
    return { ok: false, msg: 'Không đọc được sheet MEETING để chẩn đoán.', errorId: 'PERF-' + Utilities.getUuid().slice(0, 8) };
  }

  const json = JSON.stringify(meetings);
  const jsonBytes = utf8ByteLen_(json);
  timer.mark('measureBytes');

  // Xếp hạng trường theo tổng byte UTF-8.
  const totals = {};
  meetings.forEach(function (row) {
    if (!row || typeof row !== 'object') return;
    Object.keys(row).forEach(function (field) {
      const value = row[field];
      if (value == null || value === '') return;
      const bytes = utf8ByteLen_(String(value));
      if (!bytes) return;
      if (!totals[field]) totals[field] = { field: field, bytes: 0, recordsWithValue: 0 };
      totals[field].bytes += bytes;
      totals[field].recordsWithValue += 1;
    });
  });
  const dominantFields = Object.keys(totals).map(function (k) { return totals[k]; })
    .sort(function (a, b) { return (b.bytes - a.bytes) || a.field.localeCompare(b.field); })
    .slice(0, 12);
  timer.mark('rankFields');

  // Ước lượng số chunk cần thiết nếu chia theo trần 80KB.
  const estChunks = Math.max(1, Math.ceil(jsonBytes / PERF_CHUNK_MAX_BYTES_));

  const timing = timer.snapshot();
  perfLog_('getPerfDiagnostics', 'DIAG', jsonBytes, timing, jsonBytes > PERF_CACHE_VALUE_LIMIT_ ? 'payload_over_100kb' : '');

  return {
    ok: true,
    version: APP_CODE_VERSION,
    recordCount: meetings.length,
    jsonBytes: jsonBytes,
    jsonKiB: Math.round((jsonBytes / 1024) * 100) / 100,
    cacheValueLimitBytes: PERF_CACHE_VALUE_LIMIT_,
    exceedsCacheValueLimit: jsonBytes > PERF_CACHE_VALUE_LIMIT_,
    chunkMaxBytes: PERF_CHUNK_MAX_BYTES_,
    estimatedChunkCount: estChunks,
    dominantFields: dominantFields,
    timingMs: timing,
    note: 'Chẩn đoán chỉ đọc, không quét Drive, không sửa dữ liệu. Kích thước đo bằng byte UTF-8.'
  };
}

/********************* WEB APP ENTRYPOINT *********************/
function doGet(e) {
  const p = (e && e.parameter) ? e.parameter : {};
  if (p.action === 'ping') return jsonOut_({ ok: true, msg: 'LICH_CONG_TAC_HQKV8 đang chạy', at: now_() });

  // Hỗ trợ JSONP cho GitHub Pages (gọi qua ?data=...&callback=...)
  // Bảo mật: chỉ chấp nhận callback do frontend tạo theo mẫu _cb_... để giảm rủi ro XSS qua JSONP.
  if (p.data) {
    const cb = safeStr_(p.callback);
    const useJsonp = !!cb;
    try {
      if (useJsonp && !isValidJsonpCallback_(cb)) {
        return ContentService.createTextOutput('/* Invalid JSONP callback */')
          .setMimeType(ContentService.MimeType.JAVASCRIPT);
      }
      const payload = JSON.parse(p.data);
      const result = api(payload);
      if (useJsonp) return jsonpOut_(cb, result);
      return jsonOut_(result);
    } catch (err) {
      const result = { ok: false, msg: 'Lỗi xử lý yêu cầu. Vui lòng thử lại hoặc liên hệ quản trị.' };
      if (useJsonp) return jsonpOut_(cb, result);
      return jsonOut_(result);
    }
  }

  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('Quản lý lịch công tác HQKV8')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1.0, maximum-scale=1.0');
}

function doPost(e) {
  try {
    const p = parsePostPayload_(e);
    const result = api(p);
    return ContentService.createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, msg: 'Lỗi doPost: ' + err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function parsePostPayload_(e) {
  const body = (e && e.postData && e.postData.contents) || '';

  // 1) POST JSON trực tiếp từ fetch/google.script.run proxy.
  if (body) {
    try { return JSON.parse(body); } catch (ignore) {}
  }

  // 2) POST dạng form có field payload, dùng cho hidden iframe upload từ GitHub Pages/trang ngoài.
  if (e && e.parameter && e.parameter.payload) {
    try { return JSON.parse(e.parameter.payload); } catch (ignore) {}
  }

  // 3) enctype=text/plain: body thường có dạng payload={...}.
  if (body) {
    let text = String(body).trim();
    if (text.indexOf('payload=') === 0) {
      text = text.substring('payload='.length).trim();
      try { return JSON.parse(text); } catch (ignore) {}
    }

    // 4) application/x-www-form-urlencoded: payload=%7B...%7D.
    const m = text.match(/(?:^|&)payload=([^&]*)/);
    if (m && m[1]) {
      const decoded = decodeURIComponent(String(m[1]).replace(/\+/g, '%20'));
      try { return JSON.parse(decoded); } catch (ignore) {}
    }
  }

  throw new Error('Không đọc được payload POST. Vui lòng kiểm tra cách gọi upload.');
}

function jsonOut_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function isValidJsonpCallback_(cb) {
  return /^_cb_\d+[a-z0-9_]*$/i.test(safeStr_(cb));
}

function jsonpOut_(cb, obj) {
  const safeCb = safeStr_(cb);
  if (!isValidJsonpCallback_(safeCb)) {
    return ContentService.createTextOutput('/* Invalid JSONP callback */')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(safeCb + '(' + JSON.stringify(obj) + ');')
    .setMimeType(ContentService.MimeType.JAVASCRIPT);
}

function safeStr_(v) {
  return String(v === null || v === undefined ? '' : v).trim();
}

function now_() {
  return Utilities.formatDate(new Date(), CFG.TZ, 'yyyy-MM-dd HH:mm:ss');
}

function nowDate_() {
  return new Date();
}

function sha256_(s) {
  const b = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, String(s), Utilities.Charset.UTF_8);
  return b.map(x => { const v = (x < 0 ? x + 256 : x).toString(16); return v.length === 1 ? '0' + v : v; }).join('');
}

function passOk_(inputPassword, storedPassword) {
  const input = safeStr_(inputPassword);
  const stored = safeStr_(storedPassword);
  if (!input || !stored) return false;
  return input === stored || sha256_(input) === stored;
}

function isDefaultPass_(storedPassword) {
  const stored = safeStr_(storedPassword);
  return stored === '123456' || stored === 'Hqkv8@2026' || stored === sha256_('123456') || stored === '[hash]';
}

function getValues_(sheetName) {
  const s = sh_(sheetName);
  const range = s.getDataRange();
  return range.getNumRows() ? range.getValues() : [];
}

function findHeaderRow_(values, requiredHeaders) {
  requiredHeaders = requiredHeaders || [];
  for (let i = 0; i < values.length; i++) {
    const row = values[i].map(safeStr_);
    const ok = requiredHeaders.every(h => row.indexOf(h) >= 0);
    if (ok) return i;
  }
  throw new Error('Không tìm thấy dòng tiêu đề: ' + requiredHeaders.join(', '));
}

function table_(sheetName, requiredHeaders) {
  const s = sh_(sheetName);
  const range = s.getDataRange();
  const values = range.getValues();
  const displayValues = range.getDisplayValues();
  const hIdx = findHeaderRow_(values, requiredHeaders || []);
  const headers = values[hIdx].map(safeStr_);
  const rows = [];
  for (let i = hIdx + 1; i < values.length; i++) {
    const raw = values[i];
    const rawDisplay = displayValues[i] || [];
    if (raw.every(v => safeStr_(v) === '')) continue;
    const obj = {};
    const displayObj = {};
    headers.forEach((h, c) => {
      if (h) {
        obj[h] = raw[c];
        displayObj[h] = rawDisplay[c];
      }
    });
    rows.push({ rowIndex: i + 1, values: raw, displayValues: rawDisplay, obj, displayObj });
  }
  return { sheet: s, values, displayValues, headerRow: hIdx + 1, headers, rows };
}

// V160: reader read-only tối ưu cho đường nóng.
// - Chỉ đọc đến getLastRow()/getLastColumn(), tránh vùng định dạng thừa của getDataRange().
// - Chỉ gọi getDisplayValues() cho đúng cột cần giữ chuỗi hiển thị (ví dụ giờ).
// - Có thể memoize trong một lượt api(); không dùng cho dữ liệu cần đọc lại sau khi ghi.
function tableFast_(sheetName, requiredHeaders, displayHeaders, useRequestCache) {
  requiredHeaders = requiredHeaders || [];
  displayHeaders = displayHeaders || [];
  const cacheKey = [sheetName, requiredHeaders.join(','), displayHeaders.join(',')].join('|');
  if (useRequestCache && __fastTableCache_ && __fastTableCache_[cacheKey]) {
    return __fastTableCache_[cacheKey];
  }

  const s = sh_(sheetName);
  const lastRow = Math.max(Number(s.getLastRow()) || 0, 1);
  const lastCol = Math.max(Number(s.getLastColumn()) || 0, 1);
  const values = s.getRange(1, 1, lastRow, lastCol).getValues();
  const hIdx = findHeaderRow_(values, requiredHeaders);
  const headers = values[hIdx].map(safeStr_);
  const dataRowCount = Math.max(0, lastRow - hIdx - 1);
  const displayColumns = {};

  displayHeaders.forEach(function(header) {
    const col = headers.indexOf(header);
    if (col < 0 || !dataRowCount) return;
    try {
      displayColumns[header] = s.getRange(hIdx + 2, col + 1, dataRowCount, 1).getDisplayValues();
    } catch (e) {
      displayColumns[header] = [];
    }
  });

  const rows = [];
  for (let i = hIdx + 1; i < values.length; i++) {
    const raw = values[i];
    if (raw.every(function(v) { return safeStr_(v) === ''; })) continue;
    const obj = {};
    const displayObj = {};
    headers.forEach(function(h, c) {
      if (!h) return;
      obj[h] = raw[c];
      if (displayColumns[h]) {
        const offset = i - hIdx - 1;
        displayObj[h] = displayColumns[h][offset] ? displayColumns[h][offset][0] : '';
      }
    });
    rows.push({ rowIndex: i + 1, values: raw, displayValues: [], obj: obj, displayObj: displayObj });
  }

  const result = { sheet: s, values: values, displayValues: [], headerRow: hIdx + 1, headers: headers, rows: rows };
  if (useRequestCache) __fastTableCache_[cacheKey] = result;
  return result;
}

// Chỉ đọc vùng tiêu đề để append/update, không tải toàn bộ sheet.
function headerInfo_(sheetName, requiredHeaders) {
  const s = sh_(sheetName);
  const lastCol = Math.max(Number(s.getLastColumn()) || 0, 1);
  const scanRows = Math.max(1, Math.min(Number(s.getLastRow()) || 1, 50));
  const values = s.getRange(1, 1, scanRows, lastCol).getValues();
  const hIdx = findHeaderRow_(values, requiredHeaders || []);
  return { sheet: s, headerRow: hIdx + 1, headers: values[hIdx].map(safeStr_) };
}

function appendByHeaders_(sheetName, obj, requiredHeaders) {
  const h = headerInfo_(sheetName, requiredHeaders || []);
  const row = h.headers.map(function(key) {
    return Object.prototype.hasOwnProperty.call(obj, key) ? obj[key] : '';
  });
  const rowIndex = Math.max(h.sheet.getLastRow(), h.headerRow) + 1;
  h.sheet.getRange(rowIndex, 1, 1, row.length).setValues([row]);
  return rowIndex;
}

function updateByHeaders_(sheetName, rowIndex, patch, requiredHeaders) {
  const h = headerInfo_(sheetName, requiredHeaders || []);
  const changes = Object.keys(patch || {}).map(function(key) {
    return { key: key, col: h.headers.indexOf(key), value: patch[key] };
  }).filter(function(x) { return x.col >= 0; }).sort(function(a, b) { return a.col - b.col; });

  // Gộp các cột liên tiếp thành một setValues để giảm round-trip Apps Script ↔ Sheets.
  let i = 0;
  while (i < changes.length) {
    let j = i + 1;
    while (j < changes.length && changes[j].col === changes[j - 1].col + 1) j++;
    const group = changes.slice(i, j);
    h.sheet.getRange(rowIndex, group[0].col + 1, 1, group.length)
      .setValues([group.map(function(x) { return x.value; })]);
    i = j;
  }
}

function dateFromInput_(v) {
  if (v instanceof Date && !isNaN(v.getTime())) return new Date(v.getFullYear(), v.getMonth(), v.getDate());
  const s = safeStr_(v);
  if (!s) return null;
  let m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})$/);
  if (m) return new Date(Number(m[1]), Number(m[2]) - 1, Number(m[3]));
  m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
  if (m) return new Date(Number(m[3]), Number(m[2]) - 1, Number(m[1]));
  const d = new Date(s);
  if (!isNaN(d.getTime())) return new Date(d.getFullYear(), d.getMonth(), d.getDate());
  return null;
}

function fmtTime_(v, displayValue) {
  // Ưu tiên chuỗi hiển thị trong Google Sheet để tránh lỗi lệch giờ với ô định dạng TIME.
  // Google Sheets/GAS có thể đọc ô chỉ có giờ thành Date năm 1899 và bị lệch múi giờ lịch sử.
  const fromDisplay = parseTimeText_(displayValue);
  if (fromDisplay) return fromDisplay;

  if (typeof v === 'number') {
    return minutesToHHmm_(Math.round((((v % 1) + 1) % 1) * 24 * 60));
  }

  const fromText = parseTimeText_(v);
  if (fromText) return fromText;

  if (v instanceof Date && !isNaN(v.getTime())) {
    // Với time-only trong Sheet, lấy phần giờ/phút so với 00:00 cùng ngày, không format theo IANA TZ.
    const y = v.getFullYear();
    if (y <= 1901) {
      const base = new Date(v.getFullYear(), v.getMonth(), v.getDate());
      const minutes = Math.round((v.getTime() - base.getTime()) / 60000);
      if (minutes >= 0 && minutes < 24 * 60) return minutesToHHmm_(minutes);
    }
    return Utilities.formatDate(v, 'GMT+07:00', 'HH:mm');
  }

  return safeStr_(v);
}

function parseTimeText_(v) {
  const s0 = safeStr_(v);
  if (!s0) return '';
  let s = s0.toLowerCase()
    .replace(/giờ/g, 'h')
    .replace(/[：.]/g, ':')
    .replace(/\s+/g, ' ')
    .trim();
  const isPm = /\bpm\b|\bch\b/.test(s);
  const isAm = /\bam\b|\bsa\b/.test(s);
  s = s.replace(/\b(am|pm|sa|ch)\b/g, '').trim();

  let m = s.match(/^(\d{1,2})\s*[h:]\s*(\d{1,2})(?::\d{1,2})?$/);
  if (!m) m = s.match(/^(\d{1,2})(\d{2})$/);
  if (m) {
    let hh = Number(m[1]);
    const mm = Number(m[2]);
    if (isPm && hh < 12) hh += 12;
    if (isAm && hh === 12) hh = 0;
    if (hh >= 0 && hh < 24 && mm >= 0 && mm < 60) return minutesToHHmm_(hh * 60 + mm);
  }

  m = s.match(/^(\d{1,2})$/);
  if (m) {
    let hh = Number(m[1]);
    if (isPm && hh < 12) hh += 12;
    if (isAm && hh === 12) hh = 0;
    if (hh >= 0 && hh < 24) return minutesToHHmm_(hh * 60);
  }
  return '';
}

function minutesToHHmm_(minutes) {
  minutes = ((Number(minutes) || 0) % (24 * 60) + 24 * 60) % (24 * 60);
  return String(Math.floor(minutes / 60)).padStart(2, '0') + ':' + String(minutes % 60).padStart(2, '0');
}

function sessionCacheKey_(token) {
  return 'SESSION_' + safeStr_(token);
}

function sessionPropertyKey_(token) {
  return SESSION_PROPERTY_PREFIX_ + sha256_(safeStr_(token));
}

function parseSessionRaw_(raw) {
  if (!raw) return null;
  try {
    const obj = JSON.parse(raw);
    if (!obj) return null;
    if (obj.user && obj.expiresAt) return obj;
    // Tương thích session cache-only của các phiên bản cũ.
    if (obj.username || obj.id || obj.USERNAME || obj.USER_ID) {
      const now = Date.now();
      return { user: obj, createdAt: now, renewedAt: now, expiresAt: now + SESSION_TTL_MS, migratedFromLegacy: true };
    }
  } catch (e) {}
  return null;
}

function cacheSessionRaw_(token, raw, expiresAt) {
  try {
    const remainSeconds = Math.max(60, Math.floor((Number(expiresAt) - Date.now()) / 1000));
    CacheService.getScriptCache().put(
      sessionCacheKey_(token),
      raw,
      Math.min(SESSION_CACHE_TTL_SECONDS, remainSeconds)
    );
  } catch (e) {}
}

function saveSession_(token, session, persist) {
  const raw = JSON.stringify(session);
  cacheSessionRaw_(token, raw, session.expiresAt);
  if (persist !== false) {
    PropertiesService.getScriptProperties().setProperty(sessionPropertyKey_(token), raw);
  }
  return session;
}

function removeSession_(token) {
  token = safeStr_(token);
  if (!token) return;
  try { CacheService.getScriptCache().remove(sessionCacheKey_(token)); } catch (e) {}
  try { PropertiesService.getScriptProperties().deleteProperty(sessionPropertyKey_(token)); } catch (e) {}
}

function cleanupExpiredSessionsIfDue_() {
  const markerKey = 'SESSION_CLEANUP_AT_V163';
  const now = Date.now();
  const props = PropertiesService.getScriptProperties();
  let last = 0;
  try { last = Number(props.getProperty(markerKey) || 0); } catch (e) {}
  if (now - last < 6 * 60 * 60 * 1000) return;

  const lock = LockService.getScriptLock();
  let locked = false;
  try {
    locked = lock.tryLock(150);
    if (!locked) return;
    last = Number(props.getProperty(markerKey) || 0);
    if (now - last < 6 * 60 * 60 * 1000) return;

    const all = props.getProperties();
    const active = [];
    Object.keys(all).forEach(function(key) {
      if (key.indexOf(SESSION_PROPERTY_PREFIX_) !== 0) return;
      const session = parseSessionRaw_(all[key]);
      if (!session || Number(session.expiresAt || 0) <= now) {
        try { props.deleteProperty(key); } catch (e) {}
      } else {
        active.push({ key: key, expiresAt: Number(session.expiresAt || 0) });
      }
    });
    // Chặn tích lũy vô hạn nếu người dùng đóng trình duyệt mà không bấm Đăng xuất.
    active.sort(function(a, b) { return b.expiresAt - a.expiresAt; });
    active.slice(500).forEach(function(item) {
      try { props.deleteProperty(item.key); } catch (e) {}
    });
    props.setProperty(markerKey, String(now));
  } catch (e) {
    try { Logger.log('SESSION_CLEANUP_WARN ' + perfSafeErr_(e)); } catch (ignore) {}
  } finally {
    if (locked) try { lock.releaseLock(); } catch (e) {}
  }
}

function readSession_(token) {
  token = safeStr_(token);
  if (!token) return null;

  let raw = '';
  let fromPersistent = false;
  try { raw = CacheService.getScriptCache().get(sessionCacheKey_(token)) || ''; } catch (e) {}
  if (!raw) {
    try {
      raw = PropertiesService.getScriptProperties().getProperty(sessionPropertyKey_(token)) || '';
      fromPersistent = !!raw;
    } catch (e) {}
  }

  const session = parseSessionRaw_(raw);
  if (!session) return null;

  // Cache bị Google loại sớm: phục hồi từ ScriptProperties mà không đăng xuất người dùng.
  if (fromPersistent) cacheSessionRaw_(token, JSON.stringify(session), session.expiresAt);
  // Phiên cũ chỉ có trong cache: di chuyển sang kho bền vững.
  if (session.migratedFromLegacy) {
    delete session.migratedFromLegacy;
    try { saveSession_(token, session, true); } catch (e) {}
  }
  return session;
}

function requireAuth_(p) {
  const token = safeStr_(p && p.token);
  if (!token) throw appError_('AUTH_REQUIRED', 'Chưa đăng nhập. Vui lòng đăng nhập lại.', false);

  const session = readSession_(token);
  if (!session || !session.user) {
    removeSession_(token);
    throw appError_('AUTH_EXPIRED', 'Phiên đăng nhập không còn hiệu lực. Vui lòng đăng nhập lại.', false);
  }

  const now = Date.now();
  if (now > Number(session.expiresAt || 0)) {
    removeSession_(token);
    throw appError_('AUTH_EXPIRED', 'Phiên đăng nhập đã hết hạn. Vui lòng đăng nhập lại.', false);
  }

  // Sliding session: chỉ ghi gia hạn tối đa một lần mỗi 4 giờ, không làm chậm mọi API.
  const renewedAt = Number(session.renewedAt || session.createdAt || 0);
  if (now - renewedAt >= SESSION_RENEW_AFTER_MS) {
    session.renewedAt = now;
    session.expiresAt = now + SESSION_TTL_MS;
    try { saveSession_(token, session, true); } catch (e) {
      // Nếu ScriptProperties tạm lỗi, session cache hiện tại vẫn tiếp tục sử dụng.
      cacheSessionRaw_(token, JSON.stringify(session), session.expiresAt);
    }
  }
  return session.user;
}

function putSession_(token, user, options) {
  options = options || {};
  const now = Date.now();
  const ttlMs = Number(options.ttlMs) || SESSION_TTL_MS;
  const session = {
    user: user,
    createdAt: now,
    renewedAt: now,
    expiresAt: now + ttlMs,
    schemaVersion: 3
  };
  if (options.persist !== false) cleanupExpiredSessionsIfDue_();
  return saveSession_(token, session, options.persist !== false);
}

function invalidateUserAuthCache_(username) {

function getUserAuthRecord_(username) {
  username = safeStr_(username);
  if (!username) return null;
  const key = userAuthCacheKey_(username);
  try {
    const cached = CacheService.getScriptCache().get(key);
    if (cached) return JSON.parse(cached);
  } catch (e) {}

  // USER thường nhỏ; tableFast_ chỉ đọc đúng vùng có dữ liệu và không gọi getDisplayValues().
  const t = tableFast_(CFG.SH_USER, ['USERNAME'], [], false);
  const found = t.rows.find(function(r) { return safeStr_(r.obj.USERNAME) === username; });
  if (!found) return null;
  const record = { rowIndex: found.rowIndex, obj: found.obj };
  try { CacheService.getScriptCache().put(key, JSON.stringify(record), USER_AUTH_CACHE_TTL_SECONDS_); } catch (e) {}
  return record;
}

function updateLoginMetadata_(record, username, storedPassword, inputPassword) {
  if (!record || !record.rowIndex) return;
  const patch = {};
  const needsPasswordMigration = safeStr_(storedPassword) === safeStr_(inputPassword);
  if (needsPasswordMigration) patch.PASSWORD_HASH = sha256_(inputPassword);

  // LAST_LOGIN chỉ ghi tối đa một lần/30 phút/tài khoản để giảm round-trip Sheets.
  const throttleKey = 'LAST_LOGIN_V163_' + sha256_(username).slice(0, 24);
  let shouldWriteLastLogin = true;
  try { shouldWriteLastLogin = !CacheService.getScriptCache().get(throttleKey); } catch (e) {}
  if (shouldWriteLastLogin) patch.LAST_LOGIN = now_();

  if (Object.keys(patch).length) {
    try {
      updateByHeaders_(CFG.SH_USER, record.rowIndex, patch, ['USERNAME']);
      if (shouldWriteLastLogin) {
        try { CacheService.getScriptCache().put(throttleKey, '1', 30 * 60); } catch (e) {}
      }
      if (needsPasswordMigration) invalidateUserAuthCache_(username);
    } catch (e) {
      // Cập nhật metadata không được làm thất bại đăng nhập đã xác thực đúng.
      try { Logger.log('LOGIN_METADATA_WARN ' + perfSafeErr_(e)); } catch (ignore) {}
    }
  }
}

/********************* AUTH ************************************/
function login_(p) {
  const username = safeStr_(p.username);
  const password = safeStr_(p.password);
  if (!username || !password) return { ok: false, code: 'AUTH_INPUT_REQUIRED', msg: 'Vui lòng nhập tài khoản và mật khẩu.', version: APP_CODE_VERSION };
  if (isLoginBlocked_(username)) return { ok: false, code: 'AUTH_TEMP_BLOCKED', msg: 'Tài khoản tạm bị khóa do nhập sai nhiều lần. Vui lòng thử lại sau 15 phút.', version: APP_CODE_VERSION };

  const record = getUserAuthRecord_(username);
  if (!record) {
    recordLoginFail_(username);
    writeLog_('', 'LOGIN_FAIL', 'USERNAME', '', '', username, 'AUDIT', username, 'Không tồn tại user');
    return { ok: false, code: 'AUTH_BAD_CREDENTIALS', msg: 'Sai tài khoản hoặc mật khẩu.', version: APP_CODE_VERSION };
  }

  const o = record.obj || {};
  if (safeStr_(o.STATUS || 'Active') !== 'Active') {
    writeLog_('', 'LOGIN_FAIL', 'STATUS', '', 'Inactive', username, 'AUDIT', username, 'Tài khoản bị khóa');
    return { ok: false, code: 'AUTH_DISABLED', msg: 'Tài khoản đã bị vô hiệu hóa.', version: APP_CODE_VERSION };
  }

  const stored = safeStr_(o.PASSWORD_HASH);
  if (!passOk_(password, stored)) {
    recordLoginFail_(username);
    writeLog_('', 'LOGIN_FAIL', 'PASSWORD', '', '', username, 'AUDIT', username, 'Sai mật khẩu');
    return { ok: false, code: 'AUTH_BAD_CREDENTIALS', msg: 'Sai tài khoản hoặc mật khẩu.', version: APP_CODE_VERSION };
  }

  clearLoginFails_(username);
  const user = {
    id: safeStr_(o.USER_ID),
    username: safeStr_(o.USERNAME),
    fullname: safeStr_(o.FULLNAME),
    role: safeStr_(o.ROLE),
    chucvu: safeStr_(o.CHUCVU),
    donVi: safeStr_(o.DON_VI),
    email: safeStr_(o.EMAIL),
    scope: safeStr_(o.SCOPE),
    isDefaultPass: isDefaultPass_(stored)
  };

  const token = Utilities.getUuid() + '-' + Utilities.getUuid();
  putSession_(token, user, { persist: true });
  updateLoginMetadata_(record, username, stored, password);
  writeLog_('', 'LOGIN', '', '', '', username, 'AUDIT', username, 'Đăng nhập thành công');
  return { ok: true, token: token, user: user, sessionExpiresAt: Date.now() + SESSION_TTL_MS, version: APP_CODE_VERSION };
}

function logout_(p) {
  removeSession_(safeStr_(p.token));
  return { ok: true, version: APP_CODE_VERSION };
}

function changePassword_(p) {
  const user = requireAuth_(p);
  const oldPass = safeStr_(p.oldPassword);
  const newPass = safeStr_(p.newPassword);
  if (newPass.length < 8) return { ok: false, msg: 'Mật khẩu mới phải có ít nhất 8 ký tự.' };
  if (!/[A-ZÀ-Ỹ]/.test(newPass) || !/[a-zà-ỹ]/.test(newPass) || !/\d/.test(newPass)) {
    return { ok: false, msg: 'Mật khẩu nên có chữ hoa, chữ thường và số.' };
  }
  const record = getUserAuthRecord_(user.username);
  if (!record) return { ok: false, msg: 'Không tìm thấy người dùng.' };
  if (!passOk_(oldPass, record.obj.PASSWORD_HASH)) return { ok: false, msg: 'Mật khẩu cũ không đúng.' };
  updateByHeaders_(CFG.SH_USER, record.rowIndex, { PASSWORD_HASH: sha256_(newPass) }, ['USERNAME']);
  invalidateUserAuthCache_(user.username);
  writeLog_('', 'CHANGE_PASSWORD', 'PASSWORD_HASH', '', '', user.username, 'USER', user.id, 'Đổi mật khẩu');
  return { ok: true, msg: 'Đã đổi mật khẩu.' };
}

const LOGIN_FAIL_TTL = 15 * 60 * 1000;
const LOGIN_FAIL_MAX = 5;
function getLoginFails_(username) {
  const raw = PropertiesService.getScriptProperties().getProperty('LOGIN_FAIL_' + username) || '';
  const cutoff = Date.now() - LOGIN_FAIL_TTL;
  return raw.split(',').filter(Boolean).map(Number).filter(t => t > cutoff);
}
function recordLoginFail_(username) {
  const arr = getLoginFails_(username);
  arr.push(Date.now());
  PropertiesService.getScriptProperties().setProperty('LOGIN_FAIL_' + username, arr.join(','));
}
function clearLoginFails_(username) {
  PropertiesService.getScriptProperties().deleteProperty('LOGIN_FAIL_' + username);
}
function isLoginBlocked_(username) {
  return getLoginFails_(username).length >= LOGIN_FAIL_MAX;
}

// V143: SETUP chỉ được sửa thủ công trên Google Sheet (không có API ghi trong app này),
// nên cache 5 phút là an toàn và giảm hẳn số lần đọc sheet ở mỗi lượt tải trang.
function getSetupData_() {
  try {
    const cached = CacheService.getScriptCache().get('SETUP_CACHE_V1');
    if (cached) return JSON.parse(cached);
  } catch (e) {}

  t.rows.forEach(function(r) {
    if (safeStr_(r.obj.STATUS || 'Active') !== 'Active') return;
    const key = safeStr_(r.obj.CONFIG_KEY);
    const value = safeStr_(r.obj.CONFIG_VALUE);
    if (!key || !value) return;
    if (!setup[key]) setup[key] = [];
    setup[key].push({
      value: value,
      label: safeStr_(r.obj.CONFIG_LABEL) || value,
      order: Number(r.obj.SORT_ORDER) || 999,
      note: safeStr_(r.obj.GHI_CHU)
    });
  });
  Object.keys(setup).forEach(function(k) {
    setup[k].sort(function(a, b) { return a.order - b.order || a.label.localeCompare(b.label, 'vi'); });
  });
  try { CacheService.getScriptCache().put('SETUP_CACHE_V1', JSON.stringify(setup), 5 * 60); } catch (e) {}
  return setup;
}

function saveUser_(p) {
  const actor = requireAuth_(p);
  if (!hasRole_(actor, ['SuperAdmin'])) return { ok: false, msg: 'Chỉ SuperAdmin được thêm/sửa người dùng.' };
  const f = p.form || {};

  const t = table_(CFG.SH_USER, ['USERNAME']);
  const found = t.rows.find(r => safeStr_(r.obj.USERNAME) === username || safeStr_(r.obj.USER_ID) === safeStr_(f.userId));
  const userId = safeStr_(f.userId) || (found ? safeStr_(found.obj.USER_ID) : nextIdBySheet_('U', CFG.SH_USER, 'USER_ID', 4));
  const obj = {
    USER_ID: userId,

    FULLNAME: fullname,
    ROLE: safeStr_(f.role || 'Viewer'),
    CHUCVU: safeStr_(f.chucvu),
    DON_VI: safeStr_(f.donVi),
    EMAIL: safeStr_(f.email),
    PHONE: safeStr_(f.phone),
    SCOPE: safeStr_(f.scope || 'VIEW_ALL'),
    STATUS: safeStr_(f.status || 'Active'),
    CREATED_AT: found ? found.obj.CREATED_AT : now_(),
    NOTE: safeStr_(f.note)
  };
  if (safeStr_(f.password)) obj.PASSWORD_HASH = sha256_(safeStr_(f.password));
  if (found) {
    updateByHeaders_(CFG.SH_USER, found.rowIndex, obj, ['USERNAME']);
    invalidateUserAuthCache_(username);
    writeLog_('', 'UPDATE_USER', 'USER', '', username, actor.username, 'USER', userId, 'Cập nhật người dùng');
    return { ok: true, msg: 'Đã cập nhật người dùng.', userId };
  }
  obj.PASSWORD_HASH = obj.PASSWORD_HASH || sha256_('123456');
  appendByHeaders_(CFG.SH_USER, obj, ['USERNAME']);
  invalidateUserAuthCache_(username);
  writeLog_('', 'CREATE_USER', 'USER', '', username, actor.username, 'USER', userId, 'Tạo người dùng');
  return { ok: true, msg: 'Đã tạo người dùng.', userId };
}

function fileCountMap_() {
  const map = {};
  function ensure(id) {
    if (!map[id]) map[id] = { count: 0, firstUrl: '', folderId: '', folderName: '', hasFolderLink: false, seen: {} };
    return map[id];
  }
  function addFile(id, fileKey, fileUrl, folderId, folderName) {
    const item = ensure(id);
    const key = safeStr_(fileKey || fileUrl || ('file_' + item.count));
    if (key && item.seen[key]) return;
    if (key) item.seen[key] = true;
    item.count += 1;
    if (item.count === 1) item.firstUrl = safeStr_(fileUrl);
    else item.firstUrl = '';
    if (folderId && !item.folderId) item.folderId = folderId;
    if (folderName && !item.folderName) item.folderName = folderName;
  }

  try {
    const t = tableFast_(CFG.SH_FILE, ['MEETING_ID'], [], true);
    t.rows.forEach(r => {
      if (safeStr_(r.obj.STATUS || 'Active') === 'Deleted') return;
      if (normalizeFileVisibility_(r.obj.FILE_VISIBILITY) !== 'PUBLIC') return;
      const id = safeStr_(r.obj.MEETING_ID);
      if (!id) return;

      const rawUrl = safeStr_(r.obj.FILE_URL);
      const rawDriveId = safeStr_(r.obj.DRIVE_FILE_ID);
      const mimeType = safeStr_(r.obj.MIME_TYPE);
      let folderId = safeStr_(r.obj.FOLDER_ID);
      const isFolderLink = isDriveFolderUrlServer_(rawUrl) || mimeType === 'application/vnd.google-apps.folder' || (folderId && !rawDriveId && !/\/file\/d\//.test(rawUrl));

      if (isFolderLink) {
        // PHASE 1: KHÔNG quét Drive trong đường nóng (getMeetingsRaw_/bootstrap/list).
        // Trước đây gọi folderFileSummary_(folderId) → DriveApp.getFolderById().getFiles()
        // cho từng dòng thư mục, gây chậm/timeout khi có nhiều lịch. Nay chỉ ghi nhận
        // metadata thư mục từ sheet (folderId, folderName) để nút "Mở tài liệu" vẫn chạy,
        // và đánh dấu hasFolderLink=true để báo số lượng file "chưa biết" thay vì trả 0 sai.
        // Việc đếm chính xác số file trong thư mục thuộc đường chi tiết/admin, không phải đường nóng.
        folderId = folderId || driveIdFromUrlServer_(rawUrl) || rawDriveId;
        if (!folderId) return;
        const item = ensure(id);
        item.hasFolderLink = true;
        if (!item.folderId) item.folderId = folderId;
        if (!item.folderName) item.folderName = safeStr_(r.obj.FILE_NAME) || 'Thư mục tài liệu';
        return;
      }

      const key = rawDriveId || driveIdFromUrlServer_(rawUrl) || rawUrl;
      addFile(id, key, rawUrl || (rawDriveId ? driveFileUrlServer_(rawDriveId) : ''), folderId, '');
    });

    Object.keys(map).forEach(k => {
      delete map[k].seen;
      if (map[k].count !== 1) map[k].firstUrl = '';
    });
  } catch (e) {}

  return map;
}

// V43: Dự phòng hiển thị tài liệu khi MEETING_FILE chưa/không còn metadata active.
// Không thay đổi dữ liệu, chỉ đọc thư mục Drive chuẩn của lịch để tính số file và link mở thư mục.
function fillFileCountFallbackFromMeetingFolders_(map, meetingRows) {
  map = map || {};
  meetingRows = meetingRows || [];
  const maxDriveChecks = 250; // Giới hạn để tránh chậm khi danh sách lịch rất lớn.
  let checked = 0;

  function ensure(id) {
    if (!map[id]) map[id] = { count: 0, firstUrl: '', folderId: '', folderName: '' };
    return map[id];
  }

  meetingRows.forEach(function(r) {
    const id = safeStr_(r && r.obj && r.obj.MEETING_ID);
    if (!id) return;
    const o = r.obj || {};
    let folderId = safeStr_(o.FOLDER_ID) || driveIdFromUrlServer_(o.LINK_FOLDER);
    const linkFolder = safeStr_(o.LINK_FOLDER);
    if (!folderId && linkFolder) folderId = driveIdFromUrlServer_(linkFolder);
    if (!folderId) return;

    const item = ensure(id);
    if (!item.folderId) item.folderId = folderId;
    if (!item.folderName) item.folderName = safeStr_(o.NOI_DUNG) ? (id + '_Tài liệu') : ('Thư mục tài liệu ' + id);

    // Nếu đã có metadata active thì chỉ bổ sung folderId để nút mở thư mục luôn hoạt động.
    if (Number(item.count || 0) > 0) return;

    if (checked >= maxDriveChecks) return;
    checked++;
    try {
      const sum = folderFileSummary_(folderId);
      const cnt = Number(sum.count) || 0;
      if (cnt > 0) {
        item.count = cnt;
        item.firstUrl = cnt === 1 ? safeStr_(sum.firstFileUrl) : '';
        item.folderId = folderId;
        item.folderName = sum.folderName || item.folderName || ('Thư mục tài liệu ' + id);
      }
    } catch (e) {}
  });
}

// V142: Tên file trong Drive được hệ thống thêm tiền tố dạng
// LH0282_GIAY_MOI_V03_<tên file gốc>. Hàm này chỉ trả về tên gốc
// để hiển thị/thống kê đúng như người dùng upload ban đầu.
function cleanMeetingDocsReportFileName_(name) {
  let s = safeStr_(name);
  if (!s) return '';

  // Bỏ tiền tố chuẩn: LHxxxx_<NHOM>_Vxx_
  s = s.replace(/^LH\d{3,6}[_\-\s]+(?:GIAY[_\-\s]*MOI|TAI[_\-\s]*LIEU[_\-\s]*HOP|BAO[_\-\s]*CAO|PHU[_\-\s]*LUC|DINH[_\-\s]*KEM|KHAC|TAI[_\-\s]*LIEU|VAN[_\-\s]*BAN)[_\-\s]+V\d{1,3}[_\-\s]+/i, '');

  // Bỏ tiền tố dự phòng nếu còn mã lịch + phiên bản nhưng nhóm file phát sinh khác.
  s = s.replace(/^LH\d{3,6}[_\-\s]+[^_\-\s]+(?:[_\-\s]+[^_\-\s]+){0,4}[_\-\s]+V\d{1,3}[_\-\s]+/i, function(m) {
    return /^(LH\d{3,6})/i.test(m) ? '' : m;
  });

  // Bỏ riêng mã lịch nếu có.
  s = s.replace(/^LH\d{3,6}[_\-\s]+/i, '');

  // Một số tên sinh từ hệ thống có thể còn sót cụm nhóm file + phiên bản ở đầu.
  s = s.replace(/^(?:GIAY[_\-\s]*MOI|TAI[_\-\s]*LIEU[_\-\s]*HOP|BAO[_\-\s]*CAO|PHU[_\-\s]*LUC|DINH[_\-\s]*KEM|KHAC|TAI[_\-\s]*LIEU|VAN[_\-\s]*BAN)[_\-\s]+V\d{1,3}[_\-\s]+/i, '');

  return safeStr_(s).replace(/[_\s]+$/g, '').replace(/^[_\s]+/g, '');
}

function isInvitationDocFile_(f) {

  const cleanName = cleanMeetingDocsReportFileName_(f.fileDisplayName || f.displayFileName || f.fileName);
  const hasMaterial = hasMeetingDocsMaterialSignal_(cleanName);

  // Nếu tên gốc đã thể hiện rõ là báo cáo, phụ lục, tài liệu họp... thì phải tính,
  // kể cả khi file được upload trong nhóm/thư mục GIAY_MOI hoặc do Văn phòng upload.
  if (hasMaterial) return false;

  // Chỉ loại trừ khi chính tên file thể hiện rõ là giấy mời/thư mời/công văn mời.
  // V145: bỏ nhánh suy đoán theo fileType/note/sourceFolderName (nhóm file lúc upload) —
  // tên file mập mờ + lỡ chọn nhầm loại "Giấy mời" từng khiến tài liệu thật bị loại khỏi
  // thống kê, báo oan đơn vị "chưa nộp tài liệu". Mập mờ thì ưu tiên tính là tài liệu thật.
  return hasInvitationSignal_(cleanName);
}

function isOfficeUploadedFile_(f, userMap) {
  f = f || {};
  userMap = userMap || {};
  const username = safeStr_(f.uploadedByUsername || f.uploadedBy || f.UPLOADED_BY_USERNAME || f.UPLOADED_BY);
  const profile = username ? (userMap[username] || {}) : {};
  const s = [
    f.uploadedByFullname, f.uploadedByUsername, f.uploadedBy,
    profile.fullname, profile.role, profile.chucvu, profile.donVi
  ].join(' ');
  return isVanPhongText_(s);
}

function markMeetingDocsReportFiles_(files, prepUnits, userMap) {
  const assigned = !!safeStr_(prepUnits);
  return (files || []).map(function(file) {
    const f = Object.assign({}, file);
    const cleanName = cleanMeetingDocsReportFileName_(f.fileName);
    f.displayFileName = cleanName || safeStr_(f.fileName);
    f.fileDisplayName = f.displayFileName;

    const isInvitation = isInvitationDocFile_(f);
    const exclude = assigned && isInvitation;
    f.countForPrep = !exclude;
    f.excludeFromPrep = !!exclude;
    f.excludeReason = exclude
      ? 'Giấy mời/tài liệu mời thuần — không tính là tài liệu đơn vị được giao chuẩn bị'
      : '';
    return f;
  });
}

function meetingDocsReportFileItem_(f) {
  f = f || {};
  let openUrl = safeStr_(f.fileUrl);
  if (!openUrl && safeStr_(f.driveFileId)) openUrl = driveFileUrlServer_(f.driveFileId);
  if (!openUrl && safeStr_(f.folderId)) openUrl = 'https://drive.google.com/drive/folders/' + safeStr_(f.folderId);
  const rawFileName = safeStr_(f.fileName);

    fileName: rawFileName,
    displayFileName: displayFileName,
    fileDisplayName: displayFileName,
    fileType: safeStr_(f.fileType),
    fileUrl: openUrl,
    folderId: safeStr_(f.folderId),

    uploadedBy: safeStr_(f.uploadedByFullname || f.uploadedByUsername || f.uploadedBy),
    uploadedByUsername: safeStr_(f.uploadedByUsername || f.uploadedBy),
    uploadedByFullname: safeStr_(f.uploadedByFullname),
    uploadedAt: safeStr_(f.uploadedAt),
    fileVisibility: normalizeFileVisibility_(f.fileVisibility),

  };
}

function meetingDocsReportFilesMap_(meetingIds, userMap) {
  const wanted = {};
  (meetingIds || []).forEach(function(id) {
    id = safeStr_(id);
    if (id) wanted[id] = true;
  });
  const map = {};
  Object.keys(wanted).forEach(function(id) { map[id] = []; });
  if (!Object.keys(wanted).length) return map;

  try {
    const t = ensureSheetColumns_(CFG.SH_FILE, ['FILE_VISIBILITY','UPLOADED_BY_USERNAME','UPLOADED_BY_FULLNAME'], ['MEETING_ID']);
    t.rows.forEach(function(r) {
      const id = safeStr_(r.obj.MEETING_ID);
      if (!id || !wanted[id]) return;
      if (safeStr_(r.obj.STATUS || 'Active') === 'Deleted') return;

      const rawUrl = safeStr_(r.obj.FILE_URL);
      const rawDriveId = safeStr_(r.obj.DRIVE_FILE_ID);
      let folderId = safeStr_(r.obj.FOLDER_ID);
      let mimeType = safeStr_(r.obj.MIME_TYPE);
      const isFolderLink = isDriveFolderUrlServer_(rawUrl) || mimeType === 'application/vnd.google-apps.folder' || (folderId && !rawDriveId && !/\/file\/d\//.test(rawUrl));

      if (isFolderLink && !folderId) folderId = driveIdFromUrlServer_(rawUrl) || rawDriveId;
      if (isFolderLink && !mimeType) mimeType = 'application/vnd.google-apps.folder';

      const file = {
        fileRecordId: safeStr_(r.obj.FILE_RECORD_ID),
        meetingId: id,
        fileType: safeStr_(r.obj.FILE_TYPE),
        version: Number(r.obj.VERSION) || 1,
        fileName: safeStr_(r.obj.FILE_NAME),
        driveFileId: rawDriveId,
        fileUrl: rawUrl || (rawDriveId ? driveFileUrlServer_(rawDriveId) : ''),
        folderId: folderId,
        linkType: isFolderLink ? 'folder' : 'file',
        uploadedBy: safeStr_(r.obj.UPLOADED_BY_USERNAME || r.obj.UPLOADED_BY),
        uploadedByUsername: safeStr_(r.obj.UPLOADED_BY_USERNAME || r.obj.UPLOADED_BY),
        uploadedByFullname: safeStr_(r.obj.UPLOADED_BY_FULLNAME),
        uploadedAt: safeStr_(r.obj.UPLOADED_AT),
        fileVisibility: normalizeFileVisibility_(r.obj.FILE_VISIBILITY),
        fileVisibilityLabel: fileVisibilityLabel_(r.obj.FILE_VISIBILITY),
        mimeType: mimeType,
        sizeBytes: Number(r.obj.SIZE_BYTES) || 0,
        note: safeStr_(r.obj.NOTE),
        status: safeStr_(r.obj.STATUS || 'Active')
      };
      map[id].push(meetingDocsReportFileItem_(file));
    });

    Object.keys(map).forEach(function(id) {
      map[id].sort(function(a, b) {
        return safeStr_(a.fileType).localeCompare(safeStr_(b.fileType))
          || safeStr_(a.fileName).localeCompare(safeStr_(b.fileName), 'vi')
          || safeStr_(b.uploadedAt).localeCompare(safeStr_(a.uploadedAt));
      });
    });
  } catch (e) {}
  return map;
}

function getMeetingDocsReport_(p) {
  const user = requireAuth_(p);
  if (!canViewMeetingDocsReport_(user)) return { ok: false, msg: 'Tài khoản hiện tại chưa có quyền xem thống kê tài liệu họp.' };

  const t = table_(CFG.SH_MEETING, ['MEETING_ID']);
  const prepMap = meetingPrepUnitMap_();
  const uploaderUserMap = meetingDocsReportUserMap_();

  // V142: không gọi fileCountMap_(), không quét Drive và không gọi getMeetingFiles_ từng lịch.
  // Báo cáo chỉ đọc metadata MEETING_FILE một lần để tránh timeout khi có nhiều lịch/thư mục Drive.
  let meetings = t.rows.map(function(r) { return meetingRow_(r, null); })
    .filter(function(m) { return !m.isDeleted && !isTrucBanMeetingReport_(m); });

  const from = dateFromInput_(p.dateFrom);
  const to = dateFromInput_(p.dateTo);
  if (from) {
    const fromKey = dateKey_(isoDate_(from));
    meetings = meetings.filter(function(m) { return dateKey_(m.ngayKetThuc || m.ngayBatDau) >= fromKey; });
  }
  if (to) {
    const toKey = dateKey_(isoDate_(to));
    meetings = meetings.filter(function(m) { return dateKey_(m.ngayBatDau) <= toKey; });
  }

  const kw = safeStr_(p.keyword);
  if (kw) meetings = meetings.filter(function(m) { return meetingKeywordMatch_(m, kw); });
  const leader = safeStr_(p.leader);
  if (leader && leader !== 'all') meetings = meetings.filter(function(m) { return meetingLeaderMatches_(m, leader); });
  const status = safeStr_(p.status);
  if (status && status !== 'all') meetings = meetings.filter(function(m) { return m.trangThai === status; });
  if (!bool_(p.includeCancelled)) meetings = meetings.filter(function(m) { return safeStr_(m.trangThai).toUpperCase() !== 'CANCELLED'; });

  meetings.sort(function(a, b) {
    return dateKey_(a.ngayBatDau) - dateKey_(b.ngayBatDau)
      || safeStr_(a.gioBatDau).localeCompare(safeStr_(b.gioBatDau))
      || safeStr_(a.meetingId).localeCompare(safeStr_(b.meetingId));
  });

  const limit = Math.min(Math.max(Number(p.limit) || 500, 1), 1000);
  meetings = meetings.slice(0, limit);

  const filesMap = meetingDocsReportFilesMap_(meetings.map(function(m) { return m.meetingId; }), uploaderUserMap);

  const rows = meetings.map(function(m) {
    const prepUnits = safeStr_(m.donViChuanBi) || (prepMap[m.meetingId] || []).join('; ');
    let files = filesMap[m.meetingId] || [];
    files = markMeetingDocsReportFiles_(files, prepUnits, uploaderUserMap);
    const countedFiles = files.filter(function(f) { return f.countForPrep !== false; });
    const excludedFiles = files.filter(function(f) { return f.excludeFromPrep; });
    const assigned = !!safeStr_(prepUnits);
    const missing = assigned && countedFiles.length === 0;
    let docStatus = 'NO_ASSIGN';
    if (assigned && countedFiles.length > 0) docStatus = 'OK';
    if (missing) docStatus = 'MISSING';
    const isGiaoBan = /giao ban/.test(normalizeText_([m.noiDung, m.ghiChu].join(' ')));
    return {
      meetingId: m.meetingId,
      ngayBatDau: m.ngayBatDau,
      ngayKetThuc: m.ngayKetThuc,
      ngayHienThi: m.ngayHienThi,
      thu: m.thu,
      gioBatDau: m.gioBatDau,
      gioKetThuc: m.gioKetThuc,
      loaiLich: m.loaiLich,
      isGiaoBan: isGiaoBan,
      noiDung: m.noiDung,
      thanhPhan: m.thanhPhan,
      diaDiem: m.diaDiem,
      chuTri: m.chuTri,
      lanhDaoLienQuan: m.lanhDaoLienQuan,
      donViChuanBi: prepUnits,
      soVanBan: m.soVanBan,
      ghiChu: m.ghiChu,
      trangThai: m.trangThai,
      linkFolder: m.linkFolder,
      folderId: m.folderId || m.fileFolderId,
      assigned: assigned,
      missingDocs: missing,
      docStatus: docStatus,
      fileCount: countedFiles.length,
      totalFileCount: files.length,
      countedFileCount: countedFiles.length,
      excludedFileCount: excludedFiles.length,
      files: files,
      countedFiles: countedFiles,
      excludedFiles: excludedFiles
    };
  });

  const docStatus = safeStr_(p.docStatus || 'ALL').toUpperCase();
  let filteredRows = rows;
  if (docStatus === 'MISSING') filteredRows = rows.filter(function(r) { return r.docStatus === 'MISSING'; });
  if (docStatus === 'OK') filteredRows = rows.filter(function(r) { return r.docStatus === 'OK'; });
  if (docStatus === 'ASSIGNED') filteredRows = rows.filter(function(r) { return r.assigned; });
  if (docStatus === 'NO_ASSIGN') filteredRows = rows.filter(function(r) { return r.docStatus === 'NO_ASSIGN'; });

  const stats = {
    total: filteredRows.length,
    assigned: filteredRows.filter(function(r) { return r.assigned; }).length,
    withFiles: filteredRows.filter(function(r) { return r.fileCount > 0; }).length,
    missing: filteredRows.filter(function(r) { return r.missingDocs; }).length,
    noAssign: filteredRows.filter(function(r) { return !r.assigned; }).length,
    giaoBan: filteredRows.filter(function(r) { return r.isGiaoBan; }).length,
    totalFiles: filteredRows.reduce(function(s, r) { return s + Number(r.totalFileCount || r.fileCount || 0); }, 0),
    countedFiles: filteredRows.reduce(function(s, r) { return s + Number(r.countedFileCount || r.fileCount || 0); }, 0),
    excludedFiles: filteredRows.reduce(function(s, r) { return s + Number(r.excludedFileCount || 0); }, 0)
  };

  return {
    ok: true,
    rows: filteredRows,
    stats: stats,
    filters: {
      dateFrom: from ? isoDate_(from) : '',
      dateTo: to ? isoDate_(to) : '',
      keyword: kw,
      leader: leader || 'all',
      status: status || 'all',
      docStatus: docStatus,
      includeCancelled: bool_(p.includeCancelled)
    },
    note: 'Báo cáo bao gồm lịch giao ban; loại trừ lịch trực ban. Chỉ giấy mời/tài liệu mời thuần không được tính; tài liệu chuyên môn do Văn phòng upload thay đơn vị vẫn được tính nếu không phải giấy mời thuần.',
    version: APP_CODE_VERSION
  };
}

// V143: phần đọc sheet + build danh sách lịch (tốn kém nhất, có thể gọi Drive API dự phòng)
// được cache 2 lớp: (1) trong phạm vi 1 lượt gọi api() qua __meetingsRawCache_, và
// (2) qua CacheService 30 giây dùng chung cho mọi request/mọi người dùng, để tránh việc
// mở lịch/dashboard/quản lý liên tiếp phải đọc lại toàn bộ sheet MEETING + MEETING_FILE.
// Mọi hàm ghi vào 2 sheet này PHẢI gọi invalidateMeetingsCache_() để không trả dữ liệu cũ.
const MEETINGS_RAW_CACHE_KEY_ = 'MEETINGS_RAW_CACHE_V1'; // Legacy 1-value cache — chỉ giữ để dọn khi invalidate.
// PHASE 1 (Bước 3.3): cache theo manifest + nhiều chunk để vượt giới hạn ~100KB/value
// của CacheService. Manifest lưu ở khóa cố định; mỗi chunk có khóa gắn version để tự
// vô hiệu khi deploy đổi APP_CODE_VERSION. Thiếu BẤT KỲ chunk nào = MISS toàn phần
// (không trả danh sách thiếu). TTL 600s. Rebuild được bọc trong ScriptLock + double-check.
const MEETINGS_CACHE_BASE_KEY_ = 'MEETINGS_RAW_CACHE_V2';
const MEETINGS_CACHE_MANIFEST_KEY_ = 'MEETINGS_RAW_CACHE_V2::manifest';
const MEETINGS_RAW_CACHE_TTL_ = 600; // 600s dùng chung mọi request/mọi người dùng.

// PHASE 1 (race fix): token thế hệ (generation) độc lập với manifest, dùng để phát hiện
// trường hợp một lượt rebuild đọc sheet TRƯỚC khi một lượt ghi khác gọi
// invalidateMeetingsCache_(), rồi vẫn ghi cache SAU khi invalidate đã chạy — làm dữ liệu
// cũ "sống lại" tới 600s dù invalidate đã đúng. invalidateMeetingsCache_() đổi token này;
// mọi lượt rebuild phải mang theo token đã đọc lúc bắt đầu, nếu token đổi thì không ghi
// cache. Đọc cache cũng đối chiếu generationId trong manifest với token hiện hành để bắt
// cả trường hợp cache đã bị ghi đè bởi 1 rebuild cũ hơn. TTL dài hơn cache chính để không
// tự hết hạn trước cache (21600s = trần tối đa của CacheService).
const MEETINGS_CACHE_GENERATION_KEY_ = 'MEETINGS_RAW_CACHE_V2::gen';
const MEETINGS_CACHE_GENERATION_TTL_ = 21600;

function meetingsCacheGenerationRead_() {
  try { return CacheService.getScriptCache().get(MEETINGS_CACHE_GENERATION_KEY_) || ''; }
  catch (e) { return ''; }
}

function meetingsCacheGenerationBump_() {
  try {
    const token = Utilities.getUuid();
    CacheService.getScriptCache().put(MEETINGS_CACHE_GENERATION_KEY_, token, MEETINGS_CACHE_GENERATION_TTL_);
    return token;
  } catch (e) { return ''; }
}

function meetingsChunkKey_(version, index) {
  return MEETINGS_CACHE_BASE_KEY_ + '::v=' + version + '::c=' + index;
}

// Dựng danh sách lịch từ sheet (không cache). Đường nóng KHÔNG quét Drive:
// trước đây gọi fillFileCountFallbackFromMeetingFolders_() để đọc từng thư mục Drive
// (folderFileSummary_ → DriveApp.getFolderById().getFiles()) cho tối đa 250 lịch,
// gây chậm/timeout khi danh sách lớn. Nút "Mở tài liệu" vẫn hoạt động vì meetingRow_
// đọc thẳng FOLDER_ID/LINK_FOLDER từ sheet MEETING; đếm chính xác số file thuộc
// đường chi tiết/admin, không phải đường nóng.
function buildMeetingsRawFromSheet_() {
  const t = tableFast_(CFG.SH_MEETING, ['MEETING_ID'], ['GIO_BAT_DAU', 'GIO_KET_THUC'], true);
  const fileMap = fileCountMap_();
  return t.rows.map(r => meetingRow_(r, fileMap)).filter(m => !m.isDeleted);
}

// Đọc cache manifest + chunk. Trả null nếu MISS ở bất kỳ khâu nào (không manifest,
// sai schemaVersion, thiếu chunk, hỏng JSON, lệch số bản ghi, hoặc generationId đã cũ
// so với token hiện hành — xem MEETINGS_CACHE_GENERATION_KEY_) — không bao giờ trả 1 phần.
function readMeetingsCacheFromService_() {
  try {
    const cache = CacheService.getScriptCache();
    const manifestJson = cache.get(MEETINGS_CACHE_MANIFEST_KEY_);
    if (!manifestJson) return null;

    let manifest;
    try { manifest = JSON.parse(manifestJson); } catch (e) { return null; }
    if (!manifest || String(manifest.schemaVersion || '') !== String(APP_CODE_VERSION)) return null;
    if (String(manifest.generationId || '') !== String(meetingsCacheGenerationRead_())) return null;

    const keys = Array.isArray(manifest.chunkKeys) ? manifest.chunkKeys : [];
    if (keys.length !== Number(manifest.partCount)) return null;
    if (!keys.length) return Number(manifest.itemCount) === 0 ? [] : null;

    const found = cache.getAll(keys); // { key: value } chỉ cho các khóa còn sống.
    const records = [];
    for (let i = 0; i < keys.length; i++) {
      const raw = found[keys[i]];
      if (raw == null) return null; // Thiếu 1 chunk = MISS toàn phần.
      let part;
      try { part = JSON.parse(raw); } catch (e) { return null; }
      if (!Array.isArray(part)) return null;
      for (let j = 0; j < part.length; j++) records.push(part[j]);
    }
    if (records.length !== Number(manifest.itemCount)) return null;
    return records;
  } catch (e) { return null; }
}

// Ghi cache: chia bản ghi thành các chunk <= PERF_CHUNK_MAX_BYTES_ (đo byte UTF-8),
// putAll manifest + chunk cùng TTL. Nếu 1 bản ghi vượt trần chunk thì bỏ cache lần này
// (không phá request), ghi log chẩn đoán. Không nuốt lỗi im lặng: lỗi ghi sẽ được log.
// generationId phải là token đã đọc lúc BẮT ĐẦU rebuild (trước khi đọc sheet); nếu token
// hiện hành đã đổi (một invalidate đã chạy trong lúc rebuild) thì bỏ qua việc ghi để tránh
// làm "sống lại" dữ liệu cũ sau một invalidate hợp lệ.
function writeMeetingsCacheToService_(meetings, generationId) {
  try {
    if (String(generationId || '') !== String(meetingsCacheGenerationRead_())) {
      perfLog_('meetingsCacheWrite', 'SKIP', 0, null, 'generation_changed_during_rebuild');
      return;
    }

    const version = String(APP_CODE_VERSION);
    const chunkKeys = [];
    const values = {};
    let currentJson = [];
    let currentBytes = 2; // []

    const flush = function() {
      if (!currentJson.length) return;
      const key = meetingsChunkKey_(version, chunkKeys.length);
      values[key] = '[' + currentJson.join(',') + ']';
      chunkKeys.push(key);
      currentJson = [];
      currentBytes = 2;
    };

    // Mỗi record chỉ stringify một lần; tránh concat + stringify toàn chunk cho từng record
    // (O(n²)) ở bản cũ.
    for (let i = 0; i < meetings.length; i++) {
      const recordJson = JSON.stringify(meetings[i]);
      const recordBytes = utf8ByteLen_(recordJson);
      const soloBytes = recordBytes + 2;
      if (soloBytes > PERF_CHUNK_MAX_BYTES_) {
        perfLog_('meetingsCacheWrite', 'SKIP', soloBytes, null, 'single_record_exceeds_chunk_limit');
        return;
      }
      const extraBytes = recordBytes + (currentJson.length ? 1 : 0);
      if (currentJson.length && currentBytes + extraBytes > PERF_CHUNK_MAX_BYTES_) flush();
      currentJson.push(recordJson);
      currentBytes += recordBytes + (currentJson.length > 1 ? 1 : 0);
    }
    flush();

    const manifest = {
      schemaVersion: version,
      generationId: generationId,
      baseKey: MEETINGS_CACHE_BASE_KEY_,
      partCount: chunkKeys.length,
      itemCount: meetings.length,
      totalBytes: chunkKeys.reduce(function(sum, k) { return sum + utf8ByteLen_(values[k]); }, 0),
      chunkKeys: chunkKeys,
      createdAt: now_()
    };
    values[MEETINGS_CACHE_MANIFEST_KEY_] = JSON.stringify(manifest);
    CacheService.getScriptCache().putAll(values, MEETINGS_RAW_CACHE_TTL_);
  } catch (e) {
    perfLog_('meetingsCacheWrite', 'ERR', 0, null, perfSafeErr_(e));
  }
}

function getMeetingsRaw_() {
  if (__meetingsRawCache_) return __meetingsRawCache_;

  const cached = readMeetingsCacheFromService_();
  if (cached) { __meetingsRawCache_ = cached; return cached; }

  const timer = perfTimer_();
  timer.mark('cacheReadMiss');
  const lock = LockService.getScriptLock();
  let locked = false;
  try {
    // Không chờ cứng 10 giây. 4 giây đủ cho phần lớn cache rebuild; sau đó đọc cache
    // thêm một lần trước khi fallback để giao diện không có cảm giác bị treo vì lock.
    locked = lock.tryLock(4000);
    timer.mark('lockWait');
    if (locked) {
      const again = readMeetingsCacheFromService_();
      timer.mark('cacheReadAfterLock');
      if (again) { __meetingsRawCache_ = again; return again; }
    } else {
      const afterWait = readMeetingsCacheFromService_();
      timer.mark('cacheReadAfterWait');
      if (afterWait) { __meetingsRawCache_ = afterWait; return afterWait; }
    }

    let generationId = '';
    if (locked) {
      generationId = meetingsCacheGenerationRead_();
      if (!generationId) generationId = meetingsCacheGenerationBump_();
    }

    const built = buildMeetingsRawFromSheet_();
    timer.mark('rawBuild');
    __meetingsRawCache_ = built;
    if (locked) writeMeetingsCacheToService_(built, generationId);
    timer.mark('cacheWrite');

    const timing = timer.snapshot();
    const sampled = perfShouldSample_();
    let bytes = 0;
    if (perfShouldLog_(timing, '', sampled)) bytes = utf8ByteLen_(JSON.stringify(built));
    perfLog_('getMeetingsRaw', locked ? 'MISS_BUILD' : 'MISS_BUILD_NOLOCK', bytes, timing, '', sampled);
    return built;
  } catch (e) {
    perfLog_('getMeetingsRaw', 'ERR', 0, timer.snapshot(), perfSafeErr_(e), false);
    throw e;
  } finally {
    if (locked) { try { lock.releaseLock(); } catch (e) {} }
  }
}

function invalidateMeetingsCache_() {
  __meetingsRawCache_ = null;
  meetingsCacheGenerationBump_(); // Mọi rebuild đang/đã chốt token cũ sẽ không được ghi/tin cậy nữa.
  try {
    const cache = CacheService.getScriptCache();
    const toRemove = [MEETINGS_CACHE_MANIFEST_KEY_, MEETINGS_RAW_CACHE_KEY_];
    const manifestJson = cache.get(MEETINGS_CACHE_MANIFEST_KEY_);
    if (manifestJson) {
      try {
        const manifest = JSON.parse(manifestJson);
        if (manifest && Array.isArray(manifest.chunkKeys)) {
          manifest.chunkKeys.forEach(function (k) { toRemove.push(k); });
        }
      } catch (e) {}
    }
    cache.removeAll(toRemove);
  } catch (e) {}
}

function buildMeetingListFromRaw_(rawMeetings, p, user, options) {
  p = p || {};
  options = options || {};
  let meetings = (rawMeetings || []).slice();

  const showPast = options.showPast !== undefined ? !!options.showPast : bool_(p.showPast);
  if (!showPast) {
    const today = new Date(); today.setHours(0, 0, 0, 0);
    meetings = meetings.filter(function(m) {
      return !m.ngayKetThuc || dateKey_(m.ngayKetThuc) >= today.getTime();
    });
  }
  const kw = safeStr_(p.keyword);
  if (kw) meetings = meetings.filter(function(m) { return meetingKeywordMatch_(m, kw); });

  const leader = safeStr_(p.leader);
  if (leader && leader !== 'all') meetings = meetings.filter(function(m) { return meetingLeaderMatches_(m, leader); });

  const type = safeStr_(p.type);
  if (type && type !== 'all') meetings = meetings.filter(function(m) { return m.loaiLich === type; });
  const status = safeStr_(p.status);
  if (status && status !== 'all') meetings = meetings.filter(function(m) { return m.trangThai === status; });

  meetings.sort(function(a, b) {
    return dateKey_(a.ngayBatDau) - dateKey_(b.ngayBatDau)
      || safeStr_(a.gioBatDau).localeCompare(safeStr_(b.gioBatDau))
      || safeStr_(a.meetingId).localeCompare(safeStr_(b.meetingId));
  });

  const explicitLimit = options.limit !== undefined ? Number(options.limit) : Number(p.limit);
  const limit = explicitLimit > 0 ? explicitLimit : 0;
  if (limit) meetings = meetings.slice(0, limit);
  if (options.applyUserMeta !== false) applyMeetingUserMeta_(meetings, user);
  return meetings;
}

function getMeetings_(p) {
  const user = requireAuth_(p);
  const meetings = buildMeetingListFromRaw_(getMeetingsRaw_(), p, user, { applyUserMeta: true, limit: Number(p.limit) || 1000 });
  return { ok: true, meetings: meetings, user: user };
}

// Tìm kiếm thông minh: không phân biệt hoa/thường, có/không dấu, cho phép tìm theo nhiều từ khóa.
function normalizeText_(s) {
  return String(s || '')
    .toLowerCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .replace(/đ/g, 'd')
    .replace(/[^a-z0-9]+/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();
}

function meetingKeywordMatch_(m, keyword) {
  const q = normalizeText_(keyword);
  if (!q) return true;

  const haystack = normalizeText_([
    m.meetingId,
    m.legacyStt,
    m.noiDung,
    m.thanhPhan,
    m.diaDiem,
    m.chuTri,
    m.lanhDaoLienQuan,
    m.donViChuanBi,
    m.soVanBan,
    m.ghiChu,
    m.ngayHienThi,
    m.thu,
    m.loaiLich,
    m.trangThai
  ].join(' '));

  // Mỗi từ khóa người dùng nhập phải xuất hiện trong dữ liệu.
  // Ví dụ: "luat chuyen doi so" khớp với "Luật Chuyển đổi số".
  return q.split(' ').filter(Boolean).every(function(part) {
    return haystack.indexOf(part) >= 0;
  });
}

function leaderCanonical_(leader) {
  const s = normalizeText_(leader);
  const map = {
    'hung': 'pham quoc hung',
    'd c hung': 'pham quoc hung',
    'dc hung': 'pham quoc hung',
    'pham quoc hung': 'pham quoc hung',

    'duong': 'ngo tung duong',
    'd c duong': 'ngo tung duong',
    'dc duong': 'ngo tung duong',
    'ngo tung duong': 'ngo tung duong',

    'loi': 'bui ngoc loi',
    'd c loi': 'bui ngoc loi',
    'dc loi': 'bui ngoc loi',
    'bui ngoc loi': 'bui ngoc loi',

    'thang': 'nguyen canh thang',
    'd c thang': 'nguyen canh thang',
    'dc thang': 'nguyen canh thang',
    'nguyen canh thang': 'nguyen canh thang'
  };
  return map[s] || s;
}

function meetingLeaderMatches_(m, leader) {
  const target = leaderCanonical_(leader);
  if (!target) return true;

  const leaderText = normalizeText_([
    m.chuTri,
    m.lanhDaoLienQuan
  ].join(' '));

  if (!leaderText) return false;

  // Khớp chính xác theo họ tên chuẩn hóa.
  if (leaderText.indexOf(target) >= 0) return true;

  // Cho phép nhập rút gọn "Hưng/Dương/Lợi/Thắng" nhưng không dùng tìm gần đúng quá rộng.
  const lastWord = target.split(' ').filter(Boolean).pop();
  if (lastWord && ['hung', 'duong', 'loi', 'thang'].indexOf(lastWord) >= 0) {
    return leaderText.split(' ').indexOf(lastWord) >= 0;
  }

  return false;
}

// V144: Dữ liệu cho màn "Thẻ lịch lãnh đạo" (chụp ảnh gửi sếp) - gộp lịch 3 ngày
// theo từng buổi Sáng/Chiều và từng lãnh đạo, tái dùng đúng meetingLeaderMatches_
// để đảm bảo khớp 100% với cách lọc "Lãnh đạo liên quan" đang dùng ở màn Quản lý lịch.
const LEADER_CARD_LEADERS_ = [
  { key: 'hung', label: 'Đ/c Hưng' },
  { key: 'duong', label: 'Đ/c Dương' },
  { key: 'loi', label: 'Đ/c Lợi' },
  { key: 'thang', label: 'Đ/c Thắng' }
];

function sessionOfMeeting_(m) {
  const h = parseInt(String(m.gioBatDau || '').split(':')[0], 10);
  if (isNaN(h)) return 'SANG';
  return h < 12 ? 'SANG' : 'CHIEU';
}
function fullDaySessionOfMeeting_(m) {
  const h1 = parseInt(String(m.gioBatDau || '').split(':')[0], 10);
  const h2 = parseInt(String(m.gioKetThuc || '').split(':')[0], 10);
  if (!isNaN(h1) && h1 < 12 && !isNaN(h2) && h2 >= 12) return ['SANG', 'CHIEU'];
  return [sessionOfMeeting_(m)];
}

function getLeaderScheduleCard_(p) {
  requireAuth_(p);
  const startDate = dateFromInput_(p.dateFrom) || dateFromInput_(isoDate_(new Date()));
  startDate.setHours(0, 0, 0, 0);

  const days = [];
  for (let i = 0; i < 3; i++) {
    const d = new Date(startDate.getTime());
    d.setDate(d.getDate() + i);
    days.push(d);
  }
  const dayStartKeys = days.map(d => d.getTime());
  const dayEndKeys = days.map(d => {
    const e = new Date(d.getTime());
    e.setHours(23, 59, 59, 999);
    return e.getTime();
  });

  const result = days.map(d => {
    const row = { date: isoDate_(d), thu: weekdayVi_(d), sessions: { SANG: {}, CHIEU: {} } };
    LEADER_CARD_LEADERS_.forEach(ld => {
      row.sessions.SANG[ld.key] = [];
      row.sessions.CHIEU[ld.key] = [];
    });
    return row;
  });

  const meetings = getMeetingsRaw_();
  meetings.forEach(m => {
    if (safeStr_(m.trangThai).toUpperCase() === 'CANCELLED') return;
    const start = dateKey_(m.ngayBatDau);
    if (!start) return;
    const end = dateKey_(m.ngayKetThuc || m.ngayBatDau) || start;
    // Lịch có giờ bắt đầu trước 12h và giờ kết thúc từ 12h trở đi = lịch cả ngày,
    // phải hiện ở CẢ 2 buổi (không chỉ SANG) để frontend gộp ô đúng.
    const sessions = fullDaySessionOfMeeting_(m);
    for (let i = 0; i < days.length; i++) {
      if (end < dayStartKeys[i] || start > dayEndKeys[i]) continue;
      LEADER_CARD_LEADERS_.forEach(ld => {
        if (meetingLeaderMatches_(m, ld.key)) {
          sessions.forEach(session => {
            result[i].sessions[session][ld.key].push({
              meetingId: m.meetingId,
              noiDung: m.noiDung,
              diaDiem: m.diaDiem,
              fileCount: m.fileCount,
              gioBatDau: m.gioBatDau,
              gioKetThuc: m.gioKetThuc
            });
          });
        }
      });
    }
  });

  return { ok: true, days: result, leaders: LEADER_CARD_LEADERS_ };
}

  const pack = findMeeting_(id);
  if (!canEditMeeting_(user, pack.meeting)) return { ok: false, msg: 'Không có quyền sửa lịch này.' };

  const patch = formToMeetingObj_(f, id, user, false);
  patch.NGAY_BAT_DAU = start;
  patch.NGAY_KET_THUC = end;
  patch.NGAY_HIEN_THI = meetingDateDisplay_(start, end);
  patch.THU = weekdayVi_(start);

  const oldObj = pack.row.obj;
  Object.keys(patch).forEach(k => {
    if (['NGUOI_TAO', 'NGAY_TAO'].indexOf(k) >= 0) return;
    const oldVal = displayVal_(oldObj[k]);
    const newVal = displayVal_(patch[k]);
    if (oldVal !== newVal) writeLog_(id, 'UPDATE_MEETING', k, oldVal, newVal, user.username, 'MEETING', id, 'Cập nhật lịch');
  });
  updateByHeaders_(CFG.SH_MEETING, pack.row.rowIndex, patch, ['MEETING_ID']);
  syncParticipantsFromForm_(id, f, user);
  invalidateMeetingsCache_();
  return { ok: true, msg: 'Đã cập nhật lịch.', meetingId: id };

function driveIdFromUrlServer_(url) {
  const s = safeStr_(url);
  let m = s.match(/\/folders\/([a-zA-Z0-9_-]+)/); if (m) return m[1];
  m = s.match(/\/file\/d\/([a-zA-Z0-9_-]+)/); if (m) return m[1];
  m = s.match(/\/(?:document|spreadsheets|presentation)\/d\/([a-zA-Z0-9_-]+)/); if (m) return m[1];
  m = s.match(/[?&]id=([a-zA-Z0-9_-]+)/); if (m) return m[1];
  return '';
}

function isDriveFolderUrlServer_(url) {
  const s = safeStr_(url);
  return /\/folders\/[a-zA-Z0-9_-]+/.test(s);
}


function driveFileUrlServer_(id) {
  id = safeStr_(id);
  return id ? 'https://drive.google.com/file/d/' + id + '/view?usp=sharing' : '';
}

function driveFolderUrlServer_(id) {
  id = safeStr_(id);
  return id ? 'https://drive.google.com/drive/folders/' + id : '';
}

function folderFileSummary_(folderId) {
  folderId = safeStr_(folderId);
  const empty = { count: 0, firstFileUrl: '', firstFileId: '', folderName: '' };
  if (!folderId) return empty;
  const cacheKey = 'FOLDER_FILE_SUMMARY_' + folderId;
  try {
    const cached = CacheService.getScriptCache().get(cacheKey);
    if (cached) return JSON.parse(cached);
  } catch (e) {}

  const out = { count: 0, firstFileUrl: '', firstFileId: '', folderName: '' };
  try {
    const folder = DriveApp.getFolderById(folderId);
    out.folderName = folder.getName();
    const files = folder.getFiles();
    while (files.hasNext()) {
      const f = files.next();
      out.count += 1;
      if (out.count === 1) {
        out.firstFileId = f.getId();
        out.firstFileUrl = driveFileUrlServer_(f.getId());
      } else {
        out.firstFileUrl = '';
      }
    }
  } catch (e) {
    out.error = e.message;
  }
  try { CacheService.getScriptCache().put(cacheKey, JSON.stringify(out), 5 * 60); } catch (e) {}
  return out;
}

  try {

        folderId: folderId,
        linkType: 'file',
        uploadedBy: safeStr_(baseRecord.uploadedBy),
        uploadedByUsername: safeStr_(baseRecord.uploadedByUsername || baseRecord.uploadedBy),
        uploadedByFullname: safeStr_(baseRecord.uploadedByFullname),
        uploadedAt: safeStr_(baseRecord.uploadedAt),
        fileVisibility: normalizeFileVisibility_(baseRecord.fileVisibility),

        sourceFolderId: folderId,
        sourceFolderName: folder.getName()
      });
    }
  } catch (e) {}
  return items;
}


// V89: Phân quyền xem tài liệu lịch họp/Họp không giấy.
// FILE_VISIBILITY: PUBLIC | LEADER_CHICUC | LEADER_PHONGDOI_UP.
function ensureSheetColumns_(sheetName, headersToAdd, requiredHeaders) {
  const t0 = table_(sheetName, requiredHeaders || []);
  let t = t0;
  let changed = false;
  (headersToAdd || []).forEach(function(h) {
    h = safeStr_(h);
    if (!h) return;
    if (t.headers.indexOf(h) < 0) {
      t.sheet.getRange(t.headerRow, t.headers.length + 1).setValue(h);
      t.headers.push(h);
      changed = true;
    }
  });
  return changed ? table_(sheetName, requiredHeaders || []) : t0;
}

function fileVisibilityLabel_(v) {

  const text = normalizeText_([
    user.role, user.chucvu, user.fullname, profile.role, profile.chucvu, profile.fullname

function canViewMeetingFile_(user, fileObj) {
  user = user || {};

function getMeetingFiles_(meetingId, user, options) {
  options = options || {};
  const t = ensureSheetColumns_(CFG.SH_FILE, ['FILE_VISIBILITY','UPLOADED_BY_USERNAME','UPLOADED_BY_FULLNAME'], ['MEETING_ID']);
  const out = [];
  t.rows
    .filter(r => safeStr_(r.obj.MEETING_ID) === safeStr_(meetingId) && safeStr_(r.obj.STATUS || 'Active') !== 'Deleted')
    .forEach(r => {
      const rawUrl = safeStr_(r.obj.FILE_URL);
      const rawDriveId = safeStr_(r.obj.DRIVE_FILE_ID);
      let folderId = safeStr_(r.obj.FOLDER_ID);
      let mimeType = safeStr_(r.obj.MIME_TYPE);
      const isFolderLink = isDriveFolderUrlServer_(rawUrl) || mimeType === 'application/vnd.google-apps.folder' || (folderId && !rawDriveId && !/\/file\/d\//.test(rawUrl));

      if (isFolderLink && !folderId) folderId = driveIdFromUrlServer_(rawUrl) || rawDriveId;
      if (isFolderLink && !mimeType) mimeType = 'application/vnd.google-apps.folder';

      const base = {
        fileRecordId: safeStr_(r.obj.FILE_RECORD_ID),
        meetingId: safeStr_(r.obj.MEETING_ID),
        fileType: safeStr_(r.obj.FILE_TYPE),
        version: Number(r.obj.VERSION) || 1,
        fileName: safeStr_(r.obj.FILE_NAME),
        driveFileId: rawDriveId,
        fileUrl: rawUrl,
        folderId: folderId,
        linkType: isFolderLink ? 'folder' : 'file',
        uploadedBy: safeStr_(r.obj.UPLOADED_BY_USERNAME || r.obj.UPLOADED_BY),
        uploadedByUsername: safeStr_(r.obj.UPLOADED_BY_USERNAME || r.obj.UPLOADED_BY),
        uploadedByFullname: safeStr_(r.obj.UPLOADED_BY_FULLNAME),
        uploadedAt: safeStr_(r.obj.UPLOADED_AT),
        fileVisibility: normalizeFileVisibility_(r.obj.FILE_VISIBILITY),
        fileVisibilityLabel: fileVisibilityLabel_(r.obj.FILE_VISIBILITY),
        mimeType: mimeType,
        sizeBytes: Number(r.obj.SIZE_BYTES) || 0,
        note: safeStr_(r.obj.NOTE),
        status: safeStr_(r.obj.STATUS || 'Active')
      };

      // Dữ liệu import cũ thường chỉ có 01 dòng là link thư mục Drive. Để người dùng bấm mở từng file,
      // hệ thống bung danh sách file thật bên trong thư mục thành các dòng ảo trong chi tiết lịch.
      if (isFolderLink && folderId) {
        const childFiles = folderFilesToMeetingItems_(folderId, base);
        if (childFiles.length) {
          const permittedChildFiles = (options.skipPermission || !user) ? childFiles : childFiles.filter(function(f){ return canViewMeetingFile_(user, f); });
          out.push.apply(out, permittedChildFiles);
          return;
        }
      }
      if (options.skipPermission || !user || canViewMeetingFile_(user, base)) out.push(base);
    });

  // V43: nếu MEETING_FILE chưa có hoặc đã ẩn metadata, vẫn bung file thật từ thư mục chuẩn LHxxxx.
  mergeMeetingFolderFilesFallback_(meetingId, out);
  out.sort((a, b) => a.fileType.localeCompare(b.fileType) || String(a.fileName || '').localeCompare(String(b.fileName || ''), 'vi') || b.version - a.version);
  return out;
}


function fileExt_(fileName) {
  const s = safeStr_(fileName).toLowerCase();
  const p = s.lastIndexOf('.');
  return p >= 0 ? s.slice(p + 1) : '';
}

function validateUploadSecurity_(fileName, mimeType, sizeBytes) {
  const ext = fileExt_(fileName);
  if (Number(sizeBytes || 0) > MAX_UPLOAD_BYTES) throw new Error('File vượt quá giới hạn 50MB.');
  if (BLOCKED_UPLOAD_EXT.indexOf(ext) >= 0 || ALLOWED_UPLOAD_EXT.indexOf(ext) < 0) {
    throw new Error('Định dạng file chưa được phép upload. Chỉ nhận PDF, Office, ảnh, text/csv và file nén.');
  }
  const mt = safeStr_(mimeType).toLowerCase();
  if (mt.indexOf('text/html') >= 0 || mt.indexOf('javascript') >= 0 || mt.indexOf('x-msdownload') >= 0) {
    throw new Error('Loại file không được phép upload.');
  }
  return true;
}

function uploadMeetingFile_(p) {
  const uploadId = safeStr_(p && p.uploadId);
  const user = requireAuth_(p);
  const meetingId = safeStr_(p.meetingId);
  if (!meetingId) return { ok: false, msg: 'Thiếu MEETING_ID.' };
  const pack = findMeeting_(meetingId);
  if (!canUploadMeeting_(user, pack.meeting)) return { ok: false, msg: 'Không có quyền upload tài liệu cho lịch này.' };

  let base64 = safeStr_(p.base64);
  if (!base64) { const res = { ok: false, msg: 'Không có dữ liệu file.' }; putUploadStatus_(uploadId, res); return res; }
  const comma = base64.indexOf(',');
  if (comma >= 0) base64 = base64.slice(comma + 1);

  const fileType = safeStr_(p.fileType || 'TAI_LIEU_KHAC');
  const fileVisibility = normalizeFileVisibility_(p.fileVisibility || p.visibility || 'PUBLIC');
  const originalName = sanitizeFileName_(p.fileName || 'file_upload');
  const mimeType = sanitizeMimeType_(p.mimeType, originalName);
  const folders = ensureMeetingFolders_(meetingId);
  // Lưu thẳng vào thư mục gốc của lịch, không dùng thư mục con
  const targetFolder = folders.rootFolder;
  const version = nextFileVersion_(meetingId, fileType);
  const safeName = meetingId + '_' + fileType + '_V' + String(version).padStart(2, '0') + '_' + originalName;

  const bytes = Utilities.base64Decode(base64);
  validateUploadSecurity_(originalName, mimeType, bytes.length);
  let blob;
  try {
    blob = Utilities.newBlob(bytes, mimeType, safeName);
  } catch (e) {
    blob = Utilities.newBlob(bytes);
    blob.setName(safeName);
    try { blob.setContentType('application/octet-stream'); } catch (_) {}
  }
  const file = targetFolder.createFile(blob);
  // V89: vẫn cấp link xem để app mở được file trên GitHub Pages; quyền hiển thị do app/API kiểm soát.
  try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch(e) {}
  ensureSheetColumns_(CFG.SH_FILE, ['FILE_VISIBILITY','UPLOADED_BY_USERNAME','UPLOADED_BY_FULLNAME'], ['MEETING_ID']);
  const fileRecordId = nextIdBySheet_('FR', CFG.SH_FILE, 'FILE_RECORD_ID', 4);
  appendByHeaders_(CFG.SH_FILE, {
    FILE_RECORD_ID: fileRecordId,
    MEETING_ID: meetingId,
    FILE_TYPE: fileType,
    VERSION: version,
    FILE_NAME: safeName,
    DRIVE_FILE_ID: file.getId(),
    FILE_URL: 'https://drive.google.com/file/d/' + file.getId() + '/view?usp=sharing',
    FOLDER_ID: targetFolder.getId(),
    UPLOADED_BY: user.username,
    UPLOADED_BY_USERNAME: user.username,
    UPLOADED_BY_FULLNAME: safeStr_(user.fullname),
    FILE_VISIBILITY: fileVisibility,
    UPLOADED_AT: now_(),
    MIME_TYPE: mimeType,
    SIZE_BYTES: bytes.length,
    NOTE: safeStr_(p.note),
    STATUS: 'Active'
  }, ['MEETING_ID']);
  writeLog_(meetingId, 'UPLOAD_FILE', 'FILE', '', safeName, user.username, 'FILE', fileRecordId, fileType + ' · ' + fileVisibilityLabel_(fileVisibility));
  invalidateMeetingsCache_();
  const res = { ok: true, msg: 'Đã upload tài liệu.', fileRecordId, fileUrl: file.getUrl(), fileName: safeName, version, fileVisibility, fileVisibilityLabel: fileVisibilityLabel_(fileVisibility), versionCode: APP_CODE_VERSION };
  putUploadStatus_(uploadId, res);
  return res;
}

function ensureMeetingFolders_(meetingId) {
  const pack = findMeeting_(meetingId);
  let rootFolder = null;
  const oldFolderId = safeStr_(pack.meeting.folderId);
  if (oldFolderId) {
    try {
      rootFolder = DriveApp.getFolderById(oldFolderId);
      // Tự động set quyền public cho thư mục cũ
      try { rootFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch(e) {}
    } catch (e) { rootFolder = null; }
  }
  if (!rootFolder) {
    const systemRoot = getSystemRootFolder_();
    const folderName = sanitizeFileName_(meetingId + '_' + pack.meeting.noiDung).slice(0, 180);
    rootFolder = getOrCreateChildFolder_(systemRoot, folderName || meetingId);
    // Đảm bảo quyền public cho folder vừa tạo
    try { rootFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch(e) {}
    updateByHeaders_(CFG.SH_MEETING, pack.row.rowIndex, { FOLDER_ID: rootFolder.getId(), LINK_FOLDER: rootFolder.getUrl() }, ['MEETING_ID']);
  }
  // Không tạo thư mục con - file lưu thẳng vào rootFolder

    CHUONG_TRINH: rootFolder, BIEN_BAN: rootFolder, KET_LUAN: rootFolder,
    TAI_LIEU_KHAC: rootFolder
  };
  return { rootFolder, sub };
}

function getSystemRootFolder_() {
  let id = safeStr_(CFG.ROOT_FOLDER_ID) || safeStr_(PropertiesService.getScriptProperties().getProperty('ROOT_FOLDER_ID'));
  if (!id) {
    try {
      const t = table_(CFG.SH_DEPT, ['MA_DON_VI']);

      id = row ? safeStr_(row.obj.ROOT_FOLDER_ID) : '';

function getOrCreateChildFolder_(parent, name) {
  const it = parent.getFoldersByName(name);
  if (it.hasNext()) return it.next();
  const newFolder = parent.createFolder(name);
  // Tự động cấp quyền public khi tạo folder mới
  try { newFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch(e) {}
  return newFolder;
}

function nextFileVersion_(meetingId, fileType) {
  const files = getMeetingFiles_(meetingId, null, {skipPermission:true}).filter(f => f.fileType === fileType);
  return files.reduce((m, f) => Math.max(m, Number(f.version) || 0), 0) + 1;
}

function sanitizeFileName_(name) {
  return safeStr_(name || 'file_upload').replace(/[\\/:*?"<>|]/g, '_') || 'file_upload';
}

function sanitizeMimeType_(mimeType, fileName) {
  const mt = safeStr_(mimeType);
  if (/^[A-Za-z0-9!#$&^_.+-]+\/[A-Za-z0-9!#$&^_.+-]+$/.test(mt)) return mt;
  const ext = safeStr_(fileName).toLowerCase().split('.').pop();
  const map = {
    pdf:'application/pdf', doc:'application/msword', docx:'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
    xls:'application/vnd.ms-excel', xlsx:'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
    ppt:'application/vnd.ms-powerpoint', pptx:'application/vnd.openxmlformats-officedocument.presentationml.presentation',
    txt:'text/plain', csv:'text/csv', jpg:'image/jpeg', jpeg:'image/jpeg', png:'image/png', gif:'image/gif', webp:'image/webp',
    zip:'application/zip', rar:'application/vnd.rar', '7z':'application/x-7z-compressed'
  };
  return map[ext] || 'application/octet-stream';
}

function getMeetingRatingMaps_(user) {
  const myMap = {};
  const publicMap = {};
  const summary = {};
  try {
    const t = tableRatings_();
    t.rows.forEach(function(r) {
      const n = ratingObj_(r);
      if (safeStr_(n.status || 'Active') === 'Deleted') return;
      if (!n.meetingId || !(n.score >= 1 && n.score <= 5)) return;

      if (!summary[n.meetingId]) summary[n.meetingId] = { sum: 0, count: 0, latest: '', latestScore: 0 };
      summary[n.meetingId].sum += n.score;
      summary[n.meetingId].count += 1;
      const updated = safeStr_(n.updatedAt || n.createdAt);
      if (!summary[n.meetingId].latest || updated >= summary[n.meetingId].latest) {
        summary[n.meetingId].latest = updated;
        summary[n.meetingId].latestScore = n.score;
      }

      if (user && safeStr_(n.ownerUsername) === safeStr_(user.username)) {
        myMap[n.meetingId] = n;
      }
    });
    Object.keys(summary).forEach(function(id) {
      const x = summary[id];
      publicMap[id] = {
        count: x.count,
        avg: x.count ? (x.sum / x.count) : 0,
        score: x.count ? Math.round(x.sum / x.count) : 0,
        latestScore: x.latestScore
      };
    });
  } catch(e) {}
  return { my: myMap, public: publicMap };
}

      if (safeStr_(o.STATUS || 'Active') === 'Deleted') return;
      if (safeStr_(o.OWNER_USERNAME) !== safeStr_(user.username)) return;
      const meetingId = safeStr_(o.MEETING_ID);
      if (!meetingId) return;
      // Không gọi noteObj_ ở đây: noteObj_ có thể đọc nội dung từ Drive, trong khi
      // bootstrap chỉ cần đếm note theo lịch.
      map[meetingId] = (map[meetingId] || 0) + 1;
    });
  } catch (e) {}
  return map;
}

// Đọc + nhớ tạm maps đánh giá/ghi chú theo user trong phạm vi 1 lượt gọi api(). Nếu user

function getMeetingUserMetaContext_(user) {
  const key = user ? safeStr_(user.username) : '';
  if (__meetingUserMetaCache_ && __meetingUserMetaCache_.username === key) return __meetingUserMetaCache_;
  const ctx = {
    username: key,
    maps: getMeetingRatingMaps_(user),
    noteCountMap: getMyMeetingNoteCountMap_(user)
  };
  __meetingUserMetaCache_ = ctx;
  return ctx;
}

function applyMeetingUserMeta_(meetings, user) {
  const canRate = canRateMeetingPrep_(user);
  const ctx = getMeetingUserMetaContext_(user);
  const maps = ctx.maps;
  const noteCountMap = ctx.noteCountMap;

  (meetings || []).forEach(function(m) {
    const my = maps.my[m.meetingId];
    const pub = maps.public[m.meetingId];

    m.canRatePrep = canRate;
    m.myPrepRating = my ? my.score : 0;
    m.myPrepRatingId = my ? my.ratingId : '';

    // Các user khác được nhìn thấy đánh giá đã có ở chế độ chỉ xem.
    // Nếu chưa có đánh giá thì non-rater không thấy sao; rater/SuperAdmin thấy 5 sao mờ để chấm.
    m.publicPrepRating = pub ? pub.score : 0;
    m.publicPrepRatingAvg = pub ? pub.avg : 0;
    m.publicPrepRatingCount = pub ? pub.count : 0;

    const visibleScore = canRate ? (m.myPrepRating || m.publicPrepRating || 0) : (m.publicPrepRating || 0);
    m.prepRatingVisibleScore = visibleScore;
    m.showPrepRating = canRate || visibleScore > 0;

    m.myNoteCount = noteCountMap[m.meetingId] || 0;
    m.hasMyNote = m.myNoteCount > 0;
  });
  return meetings;
}

function firstRowHeaderMap_(sheet) {
  const lastCol = Math.max(sheet.getLastColumn(), 1);
  const headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(safeStr_);
  const map = {};
  headers.forEach(function(h, i) { if (h) map[h] = i; });
  return map;
}

function ensureHeaderKeys_(headerMap, required) {
  const missing = (required || []).filter(function(h) { return headerMap[h] === undefined; });
  if (missing.length) throw new Error('Thiếu cột trong MEETING_RATING: ' + missing.join(', '));
}

function setByHeaderMap_(sheet, rowIndex, headerMap, patch) {
  Object.keys(patch || {}).forEach(function(k) {
    if (headerMap[k] !== undefined) sheet.getRange(rowIndex, headerMap[k] + 1).setValue(patch[k]);
  });
}

function appendByHeaderMap_(sheet, headerMap, obj) {
  const lastCol = sheet.getLastColumn();
  const row = new Array(lastCol).fill('');
  Object.keys(obj || {}).forEach(function(k) {
    if (headerMap[k] !== undefined) row[headerMap[k]] = obj[k];
  });
  sheet.appendRow(row);
  return sheet.getLastRow();
}

function makeRatingIdFast_() {

}

function setMeetingRating_(p) {
  const user = requireAuth_(p);
  const ratingText = normalizeText_([
    safeStr_(user && user.role),
    safeStr_(user && user.username),
    safeStr_(user && user.fullname),
    safeStr_(user && user.chucvu),
    safeStr_(user && user.chucVu)
  ].join(' '));
  const allowRating = canRateMeetingPrep_(user) || /superadmin|useradmin|\badmin\b|quan tri|lanh dao chi cuc|lanhdaochicuc|lanhdaocucchi/.test(ratingText);
  if (!allowRating) {
    return { ok: false, msg: 'Tài khoản hiện tại chưa được quyền đánh giá sao.' };
  }

  const meetingId = safeStr_(p.meetingId);
  const score = Math.max(1, Math.min(5, Number(p.score) || 0));
  if (!meetingId) return { ok: false, msg: 'Thiếu mã lịch.' };
  if (!(score >= 1 && score <= 5)) return { ok: false, msg: 'Mức đánh giá phải từ 1 đến 5 sao.' };

  // Tối ưu tốc độ: không gọi findMeeting_/getMeetings_, không đọc toàn bộ sheet MEETING,
  // không trả về dữ liệu lớn. Chỉ cập nhật/ghi 01 dòng MEETING_RATING rồi trả payload nhỏ.
  const lock = LockService.getScriptLock();
  let locked = false;
  try {
    locked = lock.tryLock(5000);
    if (!locked) return { ok: false, msg: 'Hệ thống đang lưu đánh giá khác, vui lòng bấm lại sau vài giây.' };

    const s = ensureRatingSheet_();
    const h = firstRowHeaderMap_(s);
    ensureHeaderKeys_(h, ['RATING_ID','MEETING_ID','SCORE','OWNER_USERNAME','OWNER_FULLNAME','OWNER_ROLE','STATUS','CREATED_AT','UPDATED_AT','GHI_CHU']);
    const now = now_();
    const lastRow = s.getLastRow();
    const lastCol = s.getLastColumn();
    let foundRow = 0;
    let foundRatingId = '';
    let oldScore = '';

    if (lastRow >= 2) {
      // Chỉ đọc các cột cần thiết của sheet MEETING_RATING, không đọc dữ liệu lịch/cuộc họp.
      const meetingVals = s.getRange(2, h.MEETING_ID + 1, lastRow - 1, 1).getValues();
      const ownerVals = s.getRange(2, h.OWNER_USERNAME + 1, lastRow - 1, 1).getValues();
      const statusVals = s.getRange(2, h.STATUS + 1, lastRow - 1, 1).getValues();
      const scoreVals = s.getRange(2, h.SCORE + 1, lastRow - 1, 1).getValues();
      const idVals = s.getRange(2, h.RATING_ID + 1, lastRow - 1, 1).getValues();
      for (let i = 0; i < meetingVals.length; i++) {
        if (safeStr_(meetingVals[i][0]) === meetingId
            && safeStr_(ownerVals[i][0]) === safeStr_(user.username)
            && safeStr_(statusVals[i][0] || 'Active') !== 'Deleted') {
          foundRow = i + 2;
          foundRatingId = safeStr_(idVals[i][0]);
          oldScore = safeStr_(scoreVals[i][0]);
          break;
        }
      }
    }

    if (foundRow) {
      setByHeaderMap_(s, foundRow, h, {
        SCORE: score,
        OWNER_FULLNAME: user.fullname,
        OWNER_ROLE: user.role,
        STATUS: 'Active',
        UPDATED_AT: now
      });
      return { ok: true, msg: 'Đã lưu đánh giá ' + score + ' sao.', meetingId: meetingId, score: score, ratingId: foundRatingId, oldScore: oldScore, savedAt: now, lightweight: true };
    }

    const ratingId = makeRatingIdFast_();
    appendByHeaderMap_(s, h, {
      RATING_ID: ratingId,
      MEETING_ID: meetingId,
      SCORE: score,
      OWNER_USERNAME: user.username,
      OWNER_FULLNAME: user.fullname,
      OWNER_ROLE: user.role,
      STATUS: 'Active',
      CREATED_AT: now,
      UPDATED_AT: now,
      GHI_CHU: ''
    });
    return { ok: true, msg: 'Đã lưu đánh giá ' + score + ' sao.', meetingId: meetingId, score: score, ratingId: ratingId, savedAt: now, lightweight: true };
  } catch (err) {
    return { ok: false, msg: 'Không lưu được đánh giá sao: ' + (err && err.message ? err.message : String(err)) };
  } finally {
    if (locked) {
      try { lock.releaseLock(); } catch(e) {}
    }
  }
}

function noteNowDateOnly_() {
  return Utilities.formatDate(new Date(), CFG.TZ || 'Asia/Ho_Chi_Minh', 'dd/MM/yyyy');

function noteDateOnly_(v) {
  if (v instanceof Date && !isNaN(v.getTime())) {
    return Utilities.formatDate(v, CFG.TZ || 'Asia/Ho_Chi_Minh', 'dd/MM/yyyy');
  }
  const s = safeStr_(v);
  if (!s) return '';
  let m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})/);
  if (m) return String(m[1]).padStart(2, '0') + '/' + String(m[2]).padStart(2, '0') + '/' + m[3];
  m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})/);
  if (m) return String(m[3]).padStart(2, '0') + '/' + String(m[2]).padStart(2, '0') + '/' + m[1];
  const d = new Date(s);
  if (!isNaN(d.getTime())) {
    return Utilities.formatDate(d, CFG.TZ || 'Asia/Ho_Chi_Minh', 'dd/MM/yyyy');
  }
  return s.replace(/\s+\d{1,2}:\d{2}(:\d{2})?.*$/, '');
}

function ensureNoteSheet_() {
  const ss = getSS_();
  let s = ss.getSheetByName(CFG.SH_NOTE || 'MEETING_NOTE');
  const headers = [
    'NOTE_ID','MEETING_ID','TITLE','CONTENT',
    'CONTENT_FILE_ID','CONTENT_FILE_URL','CONTENT_PREVIEW',

    'NOTE_TYPE','STATUS','IS_PINNED',

    s = ss.insertSheet(CFG.SH_NOTE || 'MEETING_NOTE');
    s.getRange(1, 1, 1, headers.length).setValues([headers]);
    try {
      s.setFrozenRows(1);
      s.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#edf3f9');
      s.autoResizeColumns(1, headers.length);
    } catch (e) {}
    return s;
  }
  const lastCol = Math.max(s.getLastColumn(), 1);
  const cur = s.getRange(1, 1, 1, lastCol).getValues()[0].map(safeStr_);

  headers.forEach(function(h){
    if (cur.indexOf(h) < 0) {

      changed = true;
    }
  });
  if (changed) {
    try {
      s.getRange(1, 1, 1, s.getLastColumn()).setFontWeight('bold').setBackground('#edf3f9');
      s.autoResizeColumns(1, s.getLastColumn());
    } catch (e) {}

      CONTENT_PREVIEW: preview
    };
  }
  const folder = ensureNoteFolder_(noteId);

    file.setName(fileName);

  }
  grantPrivateNoteDriveAccess_(folder, user);
  grantPrivateNoteDriveAccess_(file, user);
  return {
    // Giữ phần xem trước trong Sheet để tìm kiếm nhanh, full text nằm trong Drive.
    CONTENT: preview,

function getNoteFileData_(p) {
  const user = requireAuth_(p);
  const noteFileId = safeStr_(p.noteFileId);
  if (!noteFileId) return { ok: false, msg: 'Thiếu mã file ghi chú.' };

  const ft = tableNoteFiles_();
  const frow = ft.rows.find(function(r){ return safeStr_(r.obj.NOTE_FILE_ID) === noteFileId; });
  if (!frow) return { ok: false, msg: 'Không tìm thấy file ghi chú.' };

  const f = noteFileObj_(frow);
  if (safeStr_(f.status || 'Active') === 'Deleted') return { ok: false, msg: 'File ghi chú đã bị xóa.' };

  const nt = tableNotes_();
  const nrow = nt.rows.find(function(r){ return safeStr_(r.obj.NOTE_ID) === f.noteId; });
  if (!nrow) return { ok: false, msg: 'Không tìm thấy ghi chú gốc.' };

  const note = noteObj_(nrow);
  if (!canReadNote_(user, note)) return { ok: false, msg: 'Không có quyền xem file đính kèm ghi chú này.' };

  const fileId = safeStr_(f.driveFileId);
  if (!fileId) return { ok: false, msg: 'File ghi chú chưa có Drive ID.' };

  const file = DriveApp.getFileById(fileId);
  const size = Number(file.getSize()) || 0;
  const maxInlineBytes = 25 * 1024 * 1024; // tránh trả payload quá lớn qua JSONP
  if (size > maxInlineBytes) {
    return {
      ok: true,
      tooLarge: true,
      msg: 'File lớn hơn 25MB, vui lòng dùng nút Mở ngoài Drive hoặc tải xuống.',
      noteFileId: f.noteFileId,
      fileName: f.fileName || file.getName(),
      mimeType: f.mimeType || file.getMimeType(),
      fileSize: size,
      openUrl: f.openUrl || file.getUrl(),
      previewUrl: f.previewUrl || ''
    };
  }

  const blob = file.getBlob();
  const mimeType = safeStr_(f.mimeType) || blob.getContentType() || file.getMimeType() || 'application/octet-stream';
  const base64 = Utilities.base64Encode(blob.getBytes());
  return {
    ok: true,
    noteFileId: f.noteFileId,
    fileName: f.fileName || file.getName(),
    mimeType: mimeType,
    fileSize: size,
    base64: base64,
    dataUrl: 'data:' + mimeType + ';base64,' + base64,
    openUrl: f.openUrl || file.getUrl(),
    previewUrl: f.previewUrl || ''
  };
}

function validateNoteAttachment_(fileName, bytes) {
  const ext = String(fileName || '').toLowerCase().split('.').pop();
  if (NOTE_ALLOWED_EXT.indexOf(ext) < 0) throw new Error('Notebook chỉ cho phép đính kèm ảnh, PDF, Word, Excel.');
  if (bytes && bytes.length > MAX_UPLOAD_BYTES) throw new Error('File đính kèm vượt quá giới hạn 50MB/file.');
  if (BLOCKED_UPLOAD_EXT.indexOf(ext) >= 0) throw new Error('Định dạng file không được phép upload.');
}

function uploadNoteFile_(p) {
  const uploadId = safeStr_(p && p.uploadId);
  try {
    const user = requireAuth_(p);
    const noteId = safeStr_(p.noteId);
    if (!noteId) { const res = { ok:false, msg:'Thiếu mã ghi chú để đính kèm file.' }; putUploadStatus_(uploadId,res); return res; }
    const nt = tableNotes_();
    const found = nt.rows.find(function(r){ return safeStr_(r.obj.NOTE_ID) === noteId; });
    if (!found) { const res = { ok:false, msg:'Không tìm thấy ghi chú.' }; putUploadStatus_(uploadId,res); return res; }
    const note = noteObj_(found);
    if (!canEditNote_(user, note)) { const res = { ok:false, msg:'Không có quyền đính kèm file vào ghi chú này.' }; putUploadStatus_(uploadId,res); return res; }

    const data = safeStr_(p.data || p.base64);
    if (!data) { const res = { ok:false, msg:'Không có dữ liệu file.' }; putUploadStatus_(uploadId,res); return res; }
    const bytes = Utilities.base64Decode(data);
    const originalName = sanitizeFileName_(p.fileName || 'note_file');
    validateNoteAttachment_(originalName, bytes);
    const mimeType = safeStr_(p.mimeType) || 'application/octet-stream';
    const folder = ensureNoteFolder_(noteId);
    const safeName = originalName;
    const file = folder.createFile(Utilities.newBlob(bytes, mimeType, safeName));
    grantPrivateNoteDriveAccess_(folder, user);
    grantPrivateNoteDriveAccess_(file, user);
    const fileId = file.getId();
    const fileUrl = file.getUrl();
    ensureNoteFileSheet_();
    const noteFileId = nextIdBySheet_('NF', CFG.SH_NOTE_FILE || 'MEETING_NOTE_FILE', 'NOTE_FILE_ID', 5);
    appendByHeaders_(CFG.SH_NOTE_FILE || 'MEETING_NOTE_FILE', {
      NOTE_FILE_ID: noteFileId,
      NOTE_ID: noteId,
      MEETING_ID: note.meetingId,
      FILE_NAME: safeName,
      MIME_TYPE: file.getMimeType(),
      FILE_SIZE: file.getSize(),
      DRIVE_FILE_ID: fileId,
      FILE_URL: fileUrl,
      OPEN_URL: fileUrl,
      PREVIEW_URL: 'https://drive.google.com/file/d/' + fileId + '/preview?rm=minimal&embedded=true',
      UPLOADED_BY: user.username,
      UPLOADED_AT: now_(),
      STATUS: 'Active',
      GHI_CHU: ''
    }, ['NOTE_FILE_ID']);
    writeLog_(note.meetingId, 'UPLOAD_NOTE_FILE', 'NOTE_FILE', '', safeName, user.username, 'NOTE', noteId, 'Đính kèm file vào ghi chú');
    const res = { ok:true, msg:'Đã đính kèm file vào ghi chú.', noteId:noteId, noteFileId:noteFileId, fileName:safeName, fileUrl:fileUrl, version:APP_CODE_VERSION };
    putUploadStatus_(uploadId, res);
    return res;
  } catch (err) {
    const res = { ok:false, msg:'Lỗi upload file ghi chú: ' + err.message, version:APP_CODE_VERSION };
    putUploadStatus_(uploadId, res);
    return res;
  }
}

function ensureNoteShareSheet_() {
  const ss = getSS_();
  const name = CFG.SH_NOTE_SHARE || 'NOTE_SHARE';
  let s = ss.getSheetByName(name);
  const headers = [
    'SHARE_ID','NOTE_ID','FROM_USERNAME','FROM_FULLNAME',
    'TO_USERNAME','TO_FULLNAME','MESSAGE','IS_READ','CREATED_AT','READ_AT','STATUS','GHI_CHU'
  ];
  if (!s) {
    s = ss.insertSheet(name);
    s.getRange(1, 1, 1, headers.length).setValues([headers]);
    try {
      s.setFrozenRows(1);
      s.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#edf3f9');
      s.autoResizeColumns(1, headers.length);
    } catch (e) {}
    return s;
  }
  const lastCol = Math.max(s.getLastColumn(), 1);
  const cur = s.getRange(1, 1, 1, lastCol).getValues()[0].map(safeStr_);
  let changed = false;
  headers.forEach(function(h){
    if (cur.indexOf(h) < 0) {
      s.getRange(1, s.getLastColumn() + 1).setValue(h);
      changed = true;
    }
  });
  if (changed) {
    try {
      s.getRange(1, 1, 1, s.getLastColumn()).setFontWeight('bold').setBackground('#edf3f9');
      s.autoResizeColumns(1, s.getLastColumn());
    } catch (e) {}
  }
  return s;
}

function searchNoteRecipients_(p) {
  const user = requireAuth_(p);
  const q = normalizeText_(safeStr_(p.q || p.keyword || p.term));
  if (!q || q.length < 1) return { ok:true, users:[] };
  const t = table_(CFG.SH_USER, ['USERNAME']);
  const out = [];
  t.rows.forEach(function(r){
    const username = safeStr_(r.obj.USERNAME);
    if (!username || username === safeStr_(user.username)) return;
    if (safeStr_(r.obj.STATUS || 'Active') !== 'Active') return;
    const fullname = safeStr_(r.obj.FULLNAME);
    const hay = normalizeText_([username, fullname, r.obj.CHUCVU, r.obj.DON_VI, r.obj.EMAIL].join(' '));
    const ok = hay.indexOf(q) >= 0 || q.split(' ').every(function(x){ return hay.indexOf(x) >= 0; });
    if (!ok) return;
    out.push({
      username: username,
      fullname: fullname,
      role: safeStr_(r.obj.ROLE),
      chucvu: safeStr_(r.obj.CHUCVU),
      donVi: safeStr_(r.obj.DON_VI),
      email: safeStr_(r.obj.EMAIL)
    });
  });
  out.sort(function(a,b){ return (a.fullname || a.username).localeCompare((b.fullname || b.username), 'vi'); });
  return { ok:true, users: out.slice(0, 12) };
}

function shareNote_(p) {
  const user = requireAuth_(p);
  ensureNoteShareSheet_();
  const noteId = safeStr_(p.noteId);
  const toUsername = safeStr_(p.toUsername || p.username);
  const message = safeStr_(p.message);
  if (!noteId) return { ok:false, msg:'Thiếu mã ghi chú.' };
  if (!toUsername) return { ok:false, msg:'Chưa chọn người nhận.' };
  const tNote = tableNotes_();
  const foundNote = tNote.rows.find(function(r){ return safeStr_(r.obj.NOTE_ID) === noteId; });
  if (!foundNote) return { ok:false, msg:'Không tìm thấy ghi chú.' };
  const note = noteObj_(foundNote);
  if (!canEditNote_(user, note)) return { ok:false, msg:'Chỉ người tạo ghi chú mới được gửi ghi chú này.' };
  if (toUsername === safeStr_(user.username)) return { ok:false, msg:'Không cần gửi ghi chú cho chính mình.' };

  const tUser = table_(CFG.SH_USER, ['USERNAME']);
  const receiver = tUser.rows.find(function(r){ return safeStr_(r.obj.USERNAME) === toUsername && safeStr_(r.obj.STATUS || 'Active') === 'Active'; });
  if (!receiver) return { ok:false, msg:'Không tìm thấy người nhận hoặc tài khoản đã bị khóa.' };

  const tShare = tableNoteShares_();
  const existed = tShare.rows.find(function(r){
    return safeStr_(r.obj.NOTE_ID) === noteId && safeStr_(r.obj.TO_USERNAME) === toUsername && safeStr_(r.obj.STATUS || 'Active') !== 'Deleted';
  });
  if (existed) {
    updateByHeaders_(CFG.SH_NOTE_SHARE || 'NOTE_SHARE', existed.rowIndex, {
      FROM_USERNAME: user.username,
      FROM_FULLNAME: user.fullname,
      TO_FULLNAME: safeStr_(receiver.obj.FULLNAME),
      MESSAGE: message,
      IS_READ: false,
      CREATED_AT: noteNowDateOnly_(),

    return { ok:true, msg:'Đã gửi lại ghi chú cho ' + safeStr_(receiver.obj.FULLNAME) + '.', shareId: safeStr_(existed.obj.SHARE_ID) };
  }

  const shareId = nextIdBySheet_('NS', CFG.SH_NOTE_SHARE || 'NOTE_SHARE', 'SHARE_ID', 5);
  appendByHeaders_(CFG.SH_NOTE_SHARE || 'NOTE_SHARE', {
    SHARE_ID: shareId,
    NOTE_ID: noteId,
    FROM_USERNAME: user.username,
    FROM_FULLNAME: user.fullname,
    TO_USERNAME: toUsername,
    TO_FULLNAME: safeStr_(receiver.obj.FULLNAME),
    MESSAGE: message,
    IS_READ: false,
    CREATED_AT: noteNowDateOnly_(),
    READ_AT: '',
    STATUS: 'Active',
    GHI_CHU: ''
  }, ['SHARE_ID']);
  writeLog_(note.meetingId, 'SHARE_NOTE', 'NOTE_SHARE', '', toUsername, user.username, 'NOTE', noteId, 'Gửi ghi chú trong app');
  return { ok:true, msg:'Đã gửi ghi chú cho ' + safeStr_(receiver.obj.FULLNAME) + '.', shareId: shareId };
}

function getNotes_(p) {
  const user = requireAuth_(p);
  const meetingId = safeStr_(p.meetingId);
  const kw = normalizeText_(safeStr_(p.keyword || p.kw));
  const includeDeleted = bool_(p.includeDeleted);

  // Đọc NOTE_SHARE đúng một lần thay vì canReadNote_/getNoteShareInfoForUser_ gọi lại
  // toàn sheet cho từng note.
  let shareRows = [];
  try { shareRows = tableNoteShares_().rows; } catch (e) { shareRows = []; }
  const shareByNote = {};
  shareRows.forEach(function(r) {
    const o = r.obj || {};
    if (safeStr_(o.TO_USERNAME) !== safeStr_(user.username)) return;
    if (safeStr_(o.STATUS || 'Active') === 'Deleted') return;
    const noteId = safeStr_(o.NOTE_ID);
    if (!noteId) return;
    shareByNote[noteId] = {
      shareId: safeStr_(o.SHARE_ID),
      fromUsername: safeStr_(o.FROM_USERNAME),
      fromFullname: safeStr_(o.FROM_FULLNAME),
      message: safeStr_(o.MESSAGE),
      isRead: bool_(o.IS_READ),
      readAt: noteDateOnly_(o.READ_AT),
      createdAt: noteDateOnly_(o.CREATED_AT)
    };
  });

  const t = tableNotes_();
  let noteRows = t.rows.filter(function(r) {
    const o = r.obj || {};
    if (!includeDeleted && safeStr_(o.STATUS || 'Active') === 'Deleted') return false;
    if (meetingId && safeStr_(o.MEETING_ID) !== meetingId) return false;
    const owned = safeStr_(o.OWNER_USERNAME) === safeStr_(user.username);
    const shared = !!shareByNote[safeStr_(o.NOTE_ID)];
    return owned || shared;
  });

  let notes = noteRows.map(noteObj_).filter(function(n) {
    if (!kw) return true;
    const hay = normalizeText_([n.noteId, n.meetingId, n.title, n.content, n.ownerFullname].join(' '));
    return hay.indexOf(kw) >= 0 || kw.split(' ').every(function(x) { return hay.indexOf(x) >= 0; });
  });

  notes.sort(function(a, b) {
    const pin = (b.isPinned ? 1 : 0) - (a.isPinned ? 1 : 0);
    if (pin) return pin;
    return safeStr_(b.updatedAt || b.createdAt).localeCompare(safeStr_(a.updatedAt || a.createdAt));
  });

  const meetingMap = getMeetingTitleMap_(notes.map(function(n) { return n.meetingId; }));

  // Đọc MEETING_NOTE_FILE đúng một lần, không N lần theo số note.
  const filesByNote = {};
  try {
    tableNoteFiles_().rows.map(noteFileObj_).forEach(function(f) {
      if (safeStr_(f.status || 'Active') === 'Deleted') return;
      if (!filesByNote[f.noteId]) filesByNote[f.noteId] = [];
      filesByNote[f.noteId].push(f);
    });
  } catch (e) {}

  notes = notes.map(function(n) {
    if (n.meetingId && meetingMap[n.meetingId]) n.meeting = meetingMap[n.meetingId];
    n.canEdit = canEditNote_(user, n);
    n.isShared = safeStr_(n.ownerUsername) !== safeStr_(user.username);
    n.shareInfo = n.isShared ? (shareByNote[n.noteId] || null) : null;
    n.shareUnread = !!(n.shareInfo && !n.shareInfo.isRead);
    n.files = filesByNote[n.noteId] || [];
    n.fileCount = n.files.length;
    return n;
  });

  const unreadSharedCount = Object.keys(shareByNote).filter(function(id) { return !shareByNote[id].isRead; }).length;
  return { ok: true, notes: notes, user: user, unreadSharedCount: unreadSharedCount };
}

function saveNote_(p) {
  const user = requireAuth_(p);
  ensureNoteSheet_();
  const f = p.form || p.note || {};
  const noteId = safeStr_(p.noteId || f.noteId);
  const meetingId = safeStr_(p.meetingId || f.meetingId);
  if (meetingId && !canCreateMeetingNote_(user, meetingId)) {
    const res = { ok: false, msg: 'Không có quyền tạo/sửa note theo lịch này.' };
    putUploadStatus_(p.uploadId, res);
    return res;
  }
  let title = safeStr_(f.title);
  const content = safeStr_(f.content);
  const isPinned = bool_(f.isPinned);
  if (!title) {
    title = meetingId ? ('Ghi chú ' + meetingId) : 'Ghi chú không tiêu đề';
  }
  if (!content) { const res = { ok: false, msg: 'Vui lòng nhập nội dung ghi chú.' }; putUploadStatus_(p.uploadId, res); return res; }

  const t = tableNotes_();
  if (noteId) {
    const found = t.rows.find(function(r){ return safeStr_(r.obj.NOTE_ID) === noteId; });
    if (!found) return { ok: false, msg: 'Không tìm thấy ghi chú.' };
    const old = noteObj_(found);
    if (!canEditNote_(user, old)) return { ok: false, msg: 'Không có quyền sửa ghi chú này.' };
    if (old.meetingId && !canCreateMeetingNote_(user, old.meetingId)) return { ok: false, msg: 'Không có quyền sửa note theo lịch này.' };
    const contentFields = saveNoteContentFields_(noteId, title, content, old.contentFileId, user);
    updateByHeaders_(CFG.SH_NOTE, found.rowIndex, Object.assign({
      MEETING_ID: meetingId,
      TITLE: title,
      NOTE_TYPE: meetingId ? 'MEETING' : 'INDEPENDENT',
      IS_PINNED: isPinned,
      STATUS: 'Active',
      UPDATED_AT: noteNowDateOnly_()
    }, contentFields), ['NOTE_ID']);
    writeLog_(meetingId, 'UPDATE_NOTE', 'NOTE', old.title, title, user.username, 'NOTE', noteId, 'Cập nhật ghi chú');
    const res = { ok: true, msg: 'Đã lưu ghi chú.', noteId: noteId, contentSavedToDrive: !!contentFields.CONTENT_FILE_ID };
    putUploadStatus_(p.uploadId, res);
    return res;
  }

  if (meetingId) {
    const existing = t.rows.find(function(r) {
      return safeStr_(r.obj.MEETING_ID) === meetingId
        && safeStr_(r.obj.OWNER_USERNAME) === user.username
        && safeStr_(r.obj.STATUS || 'Active') !== 'Deleted';
    });
    if (existing) {
      const old = noteObj_(existing);
      updateByHeaders_(CFG.SH_NOTE, existing.rowIndex, {
        MEETING_ID: meetingId,
        TITLE: title,
        CONTENT: content,
        NOTE_TYPE: 'MEETING',
        IS_PINNED: isPinned,
        STATUS: 'Active',
        UPDATED_AT: noteNowDateOnly_()
      }, ['NOTE_ID']);
      writeLog_(meetingId, 'UPDATE_NOTE', 'NOTE', old.title, title, user.username, 'NOTE', old.noteId, 'Cập nhật ghi chú theo lịch đã có');
      return { ok: true, msg: 'Đã cập nhật ghi chú.', noteId: old.noteId };
    }
  }

  const newId = nextIdBySheet_('NT', CFG.SH_NOTE, 'NOTE_ID', 5);
  const contentFields = saveNoteContentFields_(newId, title, content, '', user);
  appendByHeaders_(CFG.SH_NOTE, Object.assign({
    NOTE_ID: newId,
    MEETING_ID: meetingId,
    TITLE: title,
    OWNER_USERNAME: user.username,
    OWNER_FULLNAME: user.fullname,
    OWNER_ROLE: user.role,
    NOTE_TYPE: meetingId ? 'MEETING' : 'INDEPENDENT',
    STATUS: 'Active',
    IS_PINNED: isPinned,
    CREATED_AT: noteNowDateOnly_(),
    UPDATED_AT: noteNowDateOnly_(),
    GHI_CHU: ''
  }, contentFields), ['NOTE_ID']);
  writeLog_(meetingId, 'CREATE_NOTE', 'NOTE', '', title, user.username, 'NOTE', newId, 'Tạo ghi chú');
  const res = { ok: true, msg: 'Đã tạo ghi chú.', noteId: newId, contentSavedToDrive: !!contentFields.CONTENT_FILE_ID };
  putUploadStatus_(p.uploadId, res);
  return res;
}

function deleteNote_(p) {
  const user = requireAuth_(p);
  const noteId = safeStr_(p.noteId);
  if (!noteId) return { ok: false, msg: 'Thiếu mã ghi chú.' };
  const t = tableNotes_();
  const found = t.rows.find(function(r){ return safeStr_(r.obj.NOTE_ID) === noteId; });
  if (!found) return { ok: false, msg: 'Không tìm thấy ghi chú.' };
  const n = noteObj_(found);
  if (!canEditNote_(user, n)) {
    if (isNoteSharedToUser_(user, noteId)) return hideSharedNote_(p);
    return { ok: false, msg: 'Không có quyền xóa ghi chú này.' };
  }
  updateByHeaders_(CFG.SH_NOTE, found.rowIndex, {
    STATUS: 'Deleted',
    DELETED_AT: noteNowDateOnly_(),
    UPDATED_AT: noteNowDateOnly_()
  }, ['NOTE_ID']);
  writeLog_(n.meetingId, 'DELETE_NOTE', 'NOTE', n.title, 'Deleted', user.username, 'NOTE', noteId, 'Xóa ghi chú');
  return { ok: true, msg: 'Đã xóa ghi chú.' };
}


function makeLogId_() {
  const stamp = Utilities.formatDate(new Date(), CFG.TZ, 'yyyyMMddHHmmssSSS');
  const tail = safeStr_(Utilities.getUuid()).replace(/-/g, '').slice(0, 8).toUpperCase();
  return 'LG' + stamp + '_' + tail;
}

function nextIdBySheet_(prefix, sheetName, idHeader, width) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const h = headerInfo_(sheetName, [idHeader]);
    const col = h.headers.indexOf(idHeader);
    const lastRow = h.sheet.getLastRow();
    const count = Math.max(0, lastRow - h.headerRow);
    const ids = count ? h.sheet.getRange(h.headerRow + 1, col + 1, count, 1).getValues() : [];
    let max = 0;
    const re = new RegExp('^' + prefix.replace(/[.*+?^${}()|[\]\\]/g, '\\$&') + '(\\d+)$', 'i');
    ids.forEach(function(row) {
      const m = safeStr_(row[0]).match(re);
      if (m) max = Math.max(max, Number(m[1]) || 0);
    });
    return prefix + String(max + 1).padStart(width || 4, '0');
  } finally {
    lock.releaseLock();
  }
}

function displayVal_(v) {
  if (v instanceof Date && !isNaN(v.getTime())) return Utilities.formatDate(v, CFG.TZ, 'yyyy-MM-dd HH:mm:ss');
  return safeStr_(v);
}

function uploadStatusKey_(uploadId) {
  return 'UPLOAD_STATUS_' + safeStr_(uploadId).replace(/[^a-zA-Z0-9_-]/g, '').slice(0, 80);
}

function putUploadStatus_(uploadId, status) {
  uploadId = safeStr_(uploadId);
  if (!uploadId) return;
  const safe = Object.assign({}, status || {});
  delete safe.base64;
  safe.uploadId = uploadId;
  safe.at = now_();
  try { CacheService.getScriptCache().put(uploadStatusKey_(uploadId), JSON.stringify(safe), 30 * 60); } catch (e) {}
}

function getUploadStatus_(p) {
  // V111: không bắt buộc requireAuth ở endpoint trạng thái upload.
  // Lý do: request POST upload Excel chạy qua hidden iframe/cross-site có thể xong nhưng phiên polling bị lệch token/cache;
  // uploadId là chuỗi ngẫu nhiên đủ để nhận trạng thái ngắn hạn trong 30 phút.
  const uploadId = safeStr_(p.uploadId);
  if (!uploadId) return { ok: false, msg: 'Thiếu mã uploadId.' };
  const raw = CacheService.getScriptCache().get(uploadStatusKey_(uploadId));
  if (!raw) return { ok: true, pending: true, msg: 'Chưa có phản hồi upload từ Apps Script.', uploadId: uploadId, version: APP_CODE_VERSION };
  try {
    const status = JSON.parse(raw);
    status.pending = false;
    status.version = APP_CODE_VERSION;
    return status;
  } catch (e) {
    return { ok: false, pending: false, msg: 'Không đọc được trạng thái upload.', uploadId: uploadId, version: APP_CODE_VERSION };
  }
}

function uploadLibraryFile_(p) {
  const uploadId = safeStr_(p && p.uploadId);
  const user = requireAuth_(p);
  if (!canUploadLibrary_(user)) { const res = { ok: false, msg: 'Không có quyền upload tài liệu vào Thư viện văn bản.' }; putUploadStatus_(uploadId, res); return res; }

  const rootFolderId = safeStr_(p.rootFolderId) || LIBRARY_ROOT_FOLDER_ID;
  const folderId = safeStr_(p.folderId) || rootFolderId;
  if (!libraryIsAllowedFolder_(folderId, rootFolderId)) {
    throw new Error('Thư mục upload không thuộc phạm vi Thư viện văn bản được phép truy cập.');
  }

  let base64 = safeStr_(p.base64);
  if (!base64) { const res = { ok: false, msg: 'Không có dữ liệu file.' }; putUploadStatus_(uploadId, res); return res; }
  const comma = base64.indexOf(',');
  if (comma >= 0) base64 = base64.slice(comma + 1);

  const originalName = sanitizeFileName_(p.fileName || 'file_upload');
  const mimeType = sanitizeMimeType_(p.mimeType, originalName);
  const bytes = Utilities.base64Decode(base64);
  validateUploadSecurity_(originalName, mimeType, bytes.length);

  const targetFolder = DriveApp.getFolderById(folderId);
  let blob;
  try {
    blob = Utilities.newBlob(bytes, mimeType, originalName);
  } catch (e) {
    blob = Utilities.newBlob(bytes);
    blob.setName(originalName);
    try { blob.setContentType('application/octet-stream'); } catch (_) {}
  }

  const file = targetFolder.createFile(blob);
  try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {}

  writeLog_('', 'UPLOAD_LIBRARY_FILE', 'FILE', '', originalName, user.username, 'LIBRARY_FILE', file.getId(), 'Upload vào thư mục: ' + targetFolder.getName());

  const res = {
    ok: true,
    msg: 'Đã upload tài liệu vào thư mục Drive.',
    folder: libraryFolderToItem_(targetFolder),
    item: libraryFileToItem_(file),
    fileId: file.getId(),
    fileUrl: file.getUrl(),
    fileName: file.getName(),
    version: APP_CODE_VERSION
  };
  putUploadStatus_(uploadId, res);
  return res;
}

function librarySearchFiles_(payload) {
  payload = payload || {};
  requireAuth_(payload);
  const keyword = String(payload.keyword || '').trim();
  const rootFolderId = safeStr_(payload.rootFolderId) || LIBRARY_ROOT_FOLDER_ID;
  const max = Math.min(Number(payload.max || 100), 300);

  if (!keyword) return { ok: true, items: [] };
  if (!libraryIsAllowedFolder_(rootFolderId, LIBRARY_ROOT_FOLDER_ID)) {
    throw new Error('Thư mục tìm kiếm không thuộc phạm vi Thư viện văn bản được phép truy cập.');
  }

  const root = DriveApp.getFolderById(rootFolderId);
  const normKeyword = libraryNormalizeText_(keyword);
  const tokens = normKeyword.split(' ').filter(Boolean);
  const items = [];
  librarySearchFolderRecursive_(root, tokens, items, max, 0);

  items.sort(function(a, b) {
    if (a.type !== b.type) return a.type === 'folder' ? -1 : 1;
    return String(a.name || '').localeCompare(String(b.name || ''), 'vi');
  });

  return { ok: true, keyword: keyword, items: items, version: APP_CODE_VERSION };
}

function librarySearchFolderRecursive_(folder, tokens, items, max, depth) {
  if (items.length >= max || depth > 8) return;

  const folderName = libraryNormalizeText_(folder.getName());
  if (tokens.every(function(t){ return folderName.indexOf(t) >= 0; })) {
    items.push(libraryFolderToItem_(folder));
    if (items.length >= max) return;
  }

  const files = folder.getFiles();
  while (files.hasNext() && items.length < max) {
    const file = files.next();
    const haystack = libraryNormalizeText_(file.getName());
    if (tokens.every(function(t){ return haystack.indexOf(t) >= 0; })) {
      items.push(libraryFileToItem_(file));
    }
  }

  const subFolders = folder.getFolders();
  while (subFolders.hasNext() && items.length < max) {
    librarySearchFolderRecursive_(subFolders.next(), tokens, items, max, depth + 1);
  }
}

function libraryPreviewUrlServer_(id, mimeType) {
  id = safeStr_(id);
  mimeType = safeStr_(mimeType).toLowerCase();
  if (mimeType === 'application/vnd.google-apps.document') return 'https://docs.google.com/document/d/' + id + '/preview?rm=minimal&embedded=true';
  if (mimeType === 'application/vnd.google-apps.spreadsheet') return 'https://docs.google.com/spreadsheets/d/' + id + '/preview?rm=minimal&embedded=true';
  if (mimeType === 'application/vnd.google-apps.presentation') return 'https://docs.google.com/presentation/d/' + id + '/preview?rm=minimal&embedded=true';
  if (mimeType.indexOf('pdf') >= 0 || mimeType.indexOf('msword') >= 0 || mimeType.indexOf('wordprocessingml') >= 0 || mimeType.indexOf('excel') >= 0 || mimeType.indexOf('spreadsheet') >= 0 || mimeType.indexOf('powerpoint') >= 0 || mimeType.indexOf('presentation') >= 0) {
    return libraryGviewPreviewUrlServer_(id);
  }
  return 'https://drive.google.com/file/d/' + id + '/preview?rm=minimal&embedded=true';
}

function libraryBuildPath_(folderId, rootFolderId) {
  rootFolderId = safeStr_(rootFolderId) || LIBRARY_ROOT_FOLDER_ID;
  const path = [];
  let current = DriveApp.getFolderById(folderId);
  for (let i = 0; i < 20; i++) {
    path.push({ id: current.getId(), name: current.getName() });
    if (current.getId() === rootFolderId) break;
    const parents = current.getParents();
    if (!parents.hasNext()) break;
    current = parents.next();
  }
  return path.reverse();
}
function libraryIsAllowedFolder_(folderId, rootFolderId) {
  rootFolderId = safeStr_(rootFolderId) || LIBRARY_ROOT_FOLDER_ID;
  folderId = safeStr_(folderId);
  if (folderId === rootFolderId) return true;
  try {
    let current = DriveApp.getFolderById(folderId);
    for (let i = 0; i < 20; i++) {
      if (current.getId() === rootFolderId) return true;
      const parents = current.getParents();
      if (!parents.hasNext()) return false;
      current = parents.next();
    }
  } catch (e) {
    return false;
  }
  return false;
}
function libraryNormalizeText_(s) {
  return String(s || '')
    .toLowerCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .replace(/đ/g, 'd')
    .replace(/[^a-z0-9]+/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();
}

function ensureDutySheet_(sheetName, headers) {
  const ss = getSS_();
  let sh = ss.getSheetByName(sheetName);
  if (!sh) {
    sh = ss.insertSheet(sheetName);
    sh.getRange(1, 1, 1, headers.length).setValues([headers]);
    sh.setFrozenRows(1);
    return sh;
  }
  const range = sh.getDataRange();
  const values = range.getNumRows() ? range.getValues() : [];
  if (!values.length) {
    sh.getRange(1, 1, 1, headers.length).setValues([headers]);
    sh.setFrozenRows(1);
    return sh;
  }
  const existing = values[0].map(safeStr_);
  let changed = false;
  headers.forEach(function(h) {
    if (existing.indexOf(h) < 0) {
      sh.getRange(1, existing.length + 1).setValue(h);
      existing.push(h);
      changed = true;
    }
  });
  if (changed) sh.setFrozenRows(1);
  return sh;
}

function dutySort_(a, b) {
  return safeStr_(a.dutyDate).localeCompare(safeStr_(b.dutyDate)) ||
    dutyUnitOrder_(a.unitCode) - dutyUnitOrder_(b.unitCode) ||
    dutyPositionRank_(a.position) - dutyPositionRank_(b.position) ||
    safeStr_(a.fullname).localeCompare(safeStr_(b.fullname), 'vi');
}
function dutyDefaultRange_() {
  const d = new Date(); d.setHours(0,0,0,0);
  const start = Utilities.formatDate(d, CFG.TZ, 'yyyy-MM-dd');
  const endD = new Date(d); endD.setDate(endD.getDate() + 7);
  const end = Utilities.formatDate(endD, CFG.TZ, 'yyyy-MM-dd');
  return { startDate: start, endDate: end };
}

function isDutyAdmin_(user) {
  if (hasRole_(user, ['SuperAdmin','Admin'])) return true;
  const text = normalizeText_([user && user.role, user && user.chucvu, user && user.donVi, user && user.fullname].join(' '));
  return /van phong|chanh van phong|pho chanh van phong|lanh dao|chi cuc truong|pho chi cuc truong/.test(text);
}
function canEditDutyUnit_(user, unitCode, unitName) {
  if (isDutyAdmin_(user)) return true;
  const uText = normalizeText_(unitName || dutyUnitName_(unitCode));
  const dText = normalizeText_(user && user.donVi);
  return !!dText && !!uText && (uText.indexOf(dText) >= 0 || dText.indexOf(uText) >= 0);
}

function getDutyData_(p) {
  const user = requireAuth_(p);
  ensureDutySheets_();
  const def = dutyDefaultRange_();
  const start = dutyDateIso_(p.startDate) || def.startDate;
  const end = dutyDateIso_(p.endDate) || def.endDate;
  const t = table_(DUTY_SH_ENTRY, ['DUTY_ID','DUTY_DATE','UNIT_CODE']);
  const entries = t.rows.map(function(r) {
    return {
      dutyId: safeStr_(r.obj.DUTY_ID),
      dutyDate: dutyDateIso_(r.obj.DUTY_DATE),
      dutyWeekday: safeStr_(r.obj.DUTY_WEEKDAY) || dutyWeekday_(r.obj.DUTY_DATE),
      dutyType: safeStr_(r.obj.DUTY_TYPE),
      dutyShift: safeStr_(r.obj.DUTY_SHIFT),
      unitCode: safeStr_(r.obj.UNIT_CODE),
      unitName: safeStr_(r.obj.UNIT_NAME),
      fullname: safeStr_(r.obj.FULLNAME),
      position: safeStr_(r.obj.POSITION),
      phone: safeStr_(r.obj.PHONE),
      dutyRole: safeStr_(r.obj.DUTY_ROLE),
      note: safeStr_(r.obj.NOTE),
      status: safeStr_(r.obj.STATUS || 'DRAFT'),
      createdBy: safeStr_(r.obj.CREATED_BY),
      createdAt: safeStr_(r.obj.CREATED_AT),
      updatedBy: safeStr_(r.obj.UPDATED_BY),
      updatedAt: safeStr_(r.obj.UPDATED_AT),
      submittedBy: safeStr_(r.obj.SUBMITTED_BY),
      submittedAt: safeStr_(r.obj.SUBMITTED_AT)
    };
  }).filter(function(x) {
    if (!x.dutyDate) return false;
    if (x.dutyDate < start || x.dutyDate > end) return false;
    if (safeStr_(x.status).toUpperCase() === 'DELETED') return false;
    return true;
  }).sort(dutySort_);

  const st = table_(DUTY_SH_STATUS, ['DUTY_DATE','UNIT_CODE']);
  const statuses = st.rows.map(function(r) {
    return {
      dutyDate: dutyDateIso_(r.obj.DUTY_DATE),
      unitCode: safeStr_(r.obj.UNIT_CODE),
      unitName: safeStr_(r.obj.UNIT_NAME),
      status: safeStr_(r.obj.STATUS || 'DRAFT'),
      submittedBy: safeStr_(r.obj.SUBMITTED_BY),
      submittedAt: safeStr_(r.obj.SUBMITTED_AT),
      locked: bool_(r.obj.LOCKED),
      note: safeStr_(r.obj.NOTE)
    };
  }).filter(function(x){ return x.dutyDate >= start && x.dutyDate <= end; });

  return { ok: true, version: APP_CODE_VERSION, units: dutyUnits_(), entries: entries, statuses: statuses, canAdmin: isDutyAdmin_(user) };
}

function saveDutyEntries_(p) {
  const user = requireAuth_(p);
  ensureDutySheets_();
  const dutyDate = dutyDateIso_(p.dutyDate);
  const unitCode = safeStr_(p.unitCode);
  const unitName = safeStr_(p.unitName) || dutyUnitName_(unitCode);
  const dutyType = safeStr_(p.dutyType || 'Thứ 7/CN');
  const submit = !!p.submit;
  if (!dutyDate || !unitCode) return { ok:false, msg:'Thiếu ngày trực hoặc đơn vị.' };
  if (!canEditDutyUnit_(user, unitCode, unitName)) return { ok:false, msg:'Không có quyền cập nhật danh sách trực của đơn vị này.' };

  const t = table_(DUTY_SH_ENTRY, ['DUTY_ID','DUTY_DATE','UNIT_CODE']);
  t.rows.forEach(function(r) {
    if (dutyDateIso_(r.obj.DUTY_DATE) === dutyDate && safeStr_(r.obj.UNIT_CODE) === unitCode && safeStr_(r.obj.STATUS).toUpperCase() !== 'DELETED') {
      updateByHeaders_(DUTY_SH_ENTRY, r.rowIndex, { STATUS:'Deleted', UPDATED_BY:user.username, UPDATED_AT:now_() }, ['DUTY_ID','DUTY_DATE','UNIT_CODE']);
    }
  });

  const entries = Array.isArray(p.entries) ? p.entries : [];
  entries.forEach(function(item) {
    const fullname = safeStr_(item.fullname);
    if (!fullname) return;
    const dutyId = nextIdBySheet_('DT', DUTY_SH_ENTRY, 'DUTY_ID', 5);
    appendByHeaders_(DUTY_SH_ENTRY, {
      DUTY_ID: dutyId,
      DUTY_DATE: dutyDate,
      DUTY_WEEKDAY: dutyWeekday_(dutyDate),
      DUTY_TYPE: dutyType,
      DUTY_SHIFT: safeStr_(item.dutyShift || 'Cả ngày'),
      UNIT_CODE: unitCode,
      UNIT_NAME: unitName,
      FULLNAME: fullname,
      POSITION: safeStr_(item.position),
      PHONE: safeStr_(item.phone),
      DUTY_ROLE: safeStr_(item.dutyRole),
      NOTE: safeStr_(item.note),
      STATUS: submit ? 'SUBMITTED' : 'DRAFT',
      CREATED_BY: user.username,
      CREATED_AT: now_(),
      UPDATED_BY: user.username,
      UPDATED_AT: now_(),
      SUBMITTED_BY: submit ? user.username : '',
      SUBMITTED_AT: submit ? now_() : ''
    }, ['DUTY_ID','DUTY_DATE','UNIT_CODE']);
  });
  upsertDutyUnitStatus_(dutyDate, unitCode, unitName, submit ? 'SUBMITTED' : 'DRAFT', user.username, submit);
  writeLog_('', submit ? 'SUBMIT_DUTY' : 'SAVE_DUTY', 'DUTY_ENTRY', '', unitName + ' ' + dutyDate, user.username, 'DUTY', unitCode + '_' + dutyDate, 'Cập nhật danh sách trực');
  return { ok:true, msg: submit ? 'Đã lưu và gửi Văn phòng.' : 'Đã lưu danh sách trực.' };
}


function normalizeDutyUnitCode_(unitCode, unitName) {
  const code = safeStr_(unitCode).toUpperCase();
  const direct = dutyUnits_().find(function(u){ return u.code === code; });
  if (direct) return direct.code;

  const name = normalizeText_(unitName);
  if (!name) return '';

  const aliases = [
    { code:'CHICUC', keys:['chi cuc hqkv viii','chi cuc hai quan khu vuc viii','tru so chi cuc','hqo kv viii','hqkv viii'] },
    { code:'VANPHONG', keys:['van phong','vp'] },
    { code:'HONGAI', keys:['hon gai','cang hon gai','hqck cang hon gai'] },
    { code:'CAMPHA', keys:['cam pha','cang cam pha','hqck cang cam pha'] },
    { code:'VANGIA', keys:['van gia','cang van gia','hqck cang van gia'] },
    { code:'HOANHMO', keys:['hoanh mo','hqck hoanh mo'] },
    { code:'BPS', keys:['bac phong sinh','hqck bac phong sinh'] },
    { code:'MONGCAI', keys:['mong cai','quoc te mong cai','hqck quoc te mong cai'] },
    { code:'KSHQ_HL', keys:['khu vuc ha long','kiem soat hai quan khu vuc ha long'] },
    { code:'KSHQ_MC', keys:['khu vuc mong cai','kiem soat hai quan khu vuc mong cai'] },
    { code:'KSHQ', keys:['doi kiem soat hai quan','doi kshq','kshq'] }
  ];
  for (let i = 0; i < aliases.length; i++) {
    if (aliases[i].keys.some(function(k){ return name.indexOf(k) >= 0; })) return aliases[i].code;
  }
  const found = dutyUnits_().find(function(u){ return normalizeText_(u.name) === name || normalizeText_(u.name).indexOf(name) >= 0 || name.indexOf(normalizeText_(u.name)) >= 0; });
  return found ? found.code : '';
}

function importDutyEntries_(p) {
  const statusUploadId = safeStr_(p && (p.statusUploadId || p.commitUploadId || p.importUploadId));
  try {
    const user = requireAuth_(p);
    const res = importDutyEntriesWithUser_(user, p.rows, !!p.submit);
    res.version = APP_CODE_VERSION;
    if (statusUploadId) putUploadStatus_(statusUploadId, res);
    return res;
  } catch (err) {
    const res = { ok:false, msg:'Lỗi import danh sách trực: ' + err.message, version: APP_CODE_VERSION };
    if (statusUploadId) putUploadStatus_(statusUploadId, res);
    return res;
  }
}

function importDutyEntriesWithUser_(user, inputRows, submit) {
  ensureDutySheets_();
  const rows = Array.isArray(inputRows) ? inputRows : [];
  if (!rows.length) return { ok:false, msg:'Không có dòng dữ liệu để import.' };

  const groups = {};
  const invalid = [];
  rows.forEach(function(item, idx) {
    const dutyDate = dutyDateIso_(item.dutyDate || item.NGAY_TRUC || item.date);
    const unitCode = normalizeDutyUnitCode_(item.unitCode || item.UNIT_CODE, item.unitName || item.DON_VI || item.unit || item.truSo);
    const unitName = safeStr_(item.unitName || item.DON_VI || dutyUnitName_(unitCode));
    const fullname = safeStr_(item.fullname || item.HO_TEN || item.name);
    if (!dutyDate || !unitCode || !fullname) {
      invalid.push({ row: idx + 2, msg: 'Thiếu ngày trực/đơn vị/họ tên' });
      return;
    }
    if (!canEditDutyUnit_(user, unitCode, unitName)) {
      invalid.push({ row: idx + 2, msg: 'Không có quyền cập nhật đơn vị ' + unitName });
      return;
    }
    const key = dutyDate + '|' + unitCode;
    if (!groups[key]) groups[key] = { dutyDate: dutyDate, unitCode: unitCode, unitName: dutyUnitName_(unitCode) || unitName, dutyType: safeStr_(item.dutyType || item.DUTY_TYPE || 'Thứ 7/CN'), entries: [] };
    groups[key].entries.push({
      fullname: fullname,
      position: safeStr_(item.position || item.CHUC_VU || item.chucVu),
      phone: safeStr_(item.phone || item.SO_DIEN_THOAI || item.sdt),
      note: safeStr_(item.note || item.GHI_CHU || item.ghiChu),
      dutyShift: safeStr_(item.dutyShift || item.DUTY_SHIFT || 'Cả ngày'),
      dutyRole: safeStr_(item.dutyRole || item.DUTY_ROLE || '')
    });
  });

  const keys = Object.keys(groups);
  if (!keys.length) return { ok:false, msg:'Không có dòng hợp lệ để import.', invalid: invalid };

  const t = table_(DUTY_SH_ENTRY, ['DUTY_ID','DUTY_DATE','UNIT_CODE']);
  keys.forEach(function(key) {
    const g = groups[key];
    t.rows.forEach(function(r) {
      if (dutyDateIso_(r.obj.DUTY_DATE) === g.dutyDate && safeStr_(r.obj.UNIT_CODE) === g.unitCode && safeStr_(r.obj.STATUS).toUpperCase() !== 'DELETED') {
        updateByHeaders_(DUTY_SH_ENTRY, r.rowIndex, { STATUS:'Deleted', UPDATED_BY:user.username, UPDATED_AT:now_() }, ['DUTY_ID','DUTY_DATE','UNIT_CODE']);
      }
    });
  });

  let inserted = 0;
  keys.forEach(function(key) {
    const g = groups[key];
    g.entries.forEach(function(item) {
      const dutyId = nextIdBySheet_('DT', DUTY_SH_ENTRY, 'DUTY_ID', 5);
      appendByHeaders_(DUTY_SH_ENTRY, {
        DUTY_ID: dutyId,
        DUTY_DATE: g.dutyDate,
        DUTY_WEEKDAY: dutyWeekday_(g.dutyDate),
        DUTY_TYPE: g.dutyType,
        DUTY_SHIFT: safeStr_(item.dutyShift || 'Cả ngày'),
        UNIT_CODE: g.unitCode,
        UNIT_NAME: g.unitName,
        FULLNAME: item.fullname,
        POSITION: item.position,
        PHONE: item.phone,
        DUTY_ROLE: item.dutyRole,
        NOTE: item.note,
        STATUS: submit ? 'SUBMITTED' : 'DRAFT',
        CREATED_BY: user.username,
        CREATED_AT: now_(),
        UPDATED_BY: user.username,
        UPDATED_AT: now_(),
        SUBMITTED_BY: submit ? user.username : '',
        SUBMITTED_AT: submit ? now_() : ''
      }, ['DUTY_ID','DUTY_DATE','UNIT_CODE']);
      inserted++;
    });
    upsertDutyUnitStatus_(g.dutyDate, g.unitCode, g.unitName, submit ? 'SUBMITTED' : 'DRAFT', user.username, submit);
  });

  writeLog_('', 'IMPORT_DUTY_EXCEL', 'DUTY_ENTRY', '', inserted + ' dòng / ' + keys.length + ' nhóm ngày-đơn vị', user.username, 'DUTY', 'EXCEL_IMPORT', 'Import danh sách trực từ Excel');
  return { ok:true, msg:'Đã import ' + inserted + ' dòng danh sách trực từ Excel.', inserted: inserted, groups: keys.length, invalid: invalid };
}

function submitDutyUnit_(p) {
  const user = requireAuth_(p);
  ensureDutySheets_();
  const dutyDate = dutyDateIso_(p.dutyDate);
  const unitCode = safeStr_(p.unitCode);
  const unitName = safeStr_(p.unitName) || dutyUnitName_(unitCode);
  if (!dutyDate || !unitCode) return { ok:false, msg:'Thiếu ngày trực hoặc đơn vị.' };

}

function upsertDutyUnitStatus_(dutyDate, unitCode, unitName, status, username, submitted) {
  const t = table_(DUTY_SH_STATUS, ['DUTY_DATE','UNIT_CODE']);
  const found = t.rows.find(function(r){ return dutyDateIso_(r.obj.DUTY_DATE) === dutyDate && safeStr_(r.obj.UNIT_CODE) === unitCode; });
  const obj = {
    DUTY_DATE: dutyDate,
    UNIT_CODE: unitCode,
    UNIT_NAME: unitName,
    STATUS: status,
    SUBMITTED_BY: submitted ? username : safeStr_(found && found.obj.SUBMITTED_BY),
    SUBMITTED_AT: submitted ? now_() : safeStr_(found && found.obj.SUBMITTED_AT),
    LOCKED: false,
    UPDATED_BY: username,
    UPDATED_AT: now_()
  };
  if (found) updateByHeaders_(DUTY_SH_STATUS, found.rowIndex, obj, ['DUTY_DATE','UNIT_CODE']);
  else appendByHeaders_(DUTY_SH_STATUS, obj, ['DUTY_DATE','UNIT_CODE']);
}

  const user = requireAuth_(p);
  ensureDutySheets_();
  const dutyId = safeStr_(p.dutyId);

}

// V111: Dùng upload ticket ngắn hạn cho POST Excel qua hidden iframe.
// Cách này tránh lỗi token/session khi form POST chạy qua redirect của Google Apps Script.
function createDutyUploadTicket_(p) {
  const user = requireAuth_(p);
  const ticket = 'dutytk_' + Utilities.getUuid().replace(/-/g, '') + '_' + Utilities.getUuid().replace(/-/g, '').slice(0, 8);
  const obj = {
    username: user.username,
    user: user,
    purpose: safeStr_(p.purpose || 'DUTY_EXCEL'),
    fileName: safeStr_(p.fileName),
    createdAt: Date.now(),
    expiresAt: Date.now() + 10 * 60 * 1000
  };
  CacheService.getScriptCache().put('DUTY_UPLOAD_TICKET_' + ticket, JSON.stringify(obj), 10 * 60);
  return { ok: true, ticket: ticket, expiresIn: 600, version: APP_CODE_VERSION };
}

function dutyUserFromUploadTicket_(ticket) {
  ticket = safeStr_(ticket);
  if (!ticket) return null;
  const raw = CacheService.getScriptCache().get('DUTY_UPLOAD_TICKET_' + ticket);
  if (!raw) return null;
  try {
    const obj = JSON.parse(raw);
    if (obj.expiresAt && Date.now() > Number(obj.expiresAt)) return null;
    return obj.user || null;
  } catch (e) {
    return null;
  }
}

function requireDutyUploadUser_(p) {
  const byTicket = dutyUserFromUploadTicket_(p && p.uploadTicket);
  if (byTicket && safeStr_(byTicket.username || byTicket.id)) return byTicket;
  return requireAuth_(p);
}

// Hỗ trợ file .xlsx và .csv; sheet ưu tiên DUTY_IMPORT, nếu không có thì đọc sheet đầu tiên.
function parseDutyExcelFile_(p) {
  const uploadId = safeStr_(p.uploadId);
  try {
    const user = requireDutyUploadUser_(p);
    ensureDutySheets_();
    const rawRows = dutyParseUploadedSpreadsheet_(p);
    const pack = dutyNormalizeServerImportRows_(rawRows, user, false);
    const res = { ok: true, msg: 'Đã đọc ' + pack.rows.length + ' dòng hợp lệ từ Excel.', rows: pack.rows, invalid: pack.invalid, total: rawRows.length, version: APP_CODE_VERSION };
    if (uploadId) putUploadStatus_(uploadId, res);
    return res;
  } catch (err) {
    const res = { ok: false, msg: 'Lỗi đọc Excel trên server: ' + err.message, version: APP_CODE_VERSION };
    if (uploadId) putUploadStatus_(uploadId, res);
    return res;
  }
}

function importDutyExcelFile_(p) {
  const uploadId = safeStr_(p.uploadId);

    const user = requireDutyUploadUser_(p);
    ensureDutySheets_();

    const res = { ok: false, msg: 'Lỗi import Excel trên server: ' + err.message, version: APP_CODE_VERSION };
    if (uploadId) putUploadStatus_(uploadId, res);
    return res;
  }
}


// V113: sau khi đã đọc Excel thành công, khi bấm “Cập nhật vào phần mềm”
// không upload lại file Excel lần 2. Frontend gửi uploadId của lần đọc trước,
// server lấy các dòng đã parse trong CacheService rồi ghi vào DUTY_ENTRY.
// Cách này tránh lỗi hidden iframe POST không trả callback / bị chặn mạng / hết hạn ticket.
function commitDutyExcelImport_(p) {
  const statusUploadId = safeStr_(p && (p.statusUploadId || p.commitUploadId || p.importUploadId));
  try {
    const user = requireAuth_(p);
    const uploadId = safeStr_(p.uploadId || p.parseUploadId);
    if (!uploadId) {
      const res0 = { ok: false, msg: 'Thiếu mã dữ liệu Excel đã đọc. Vui lòng bấm Đọc file Excel lại.', version: APP_CODE_VERSION };
      if (statusUploadId) putUploadStatus_(statusUploadId, res0);
      return res0;
    }
    const raw = CacheService.getScriptCache().get(uploadStatusKey_(uploadId));
    if (!raw) {
      const res0 = { ok: false, msg: 'Dữ liệu Excel đã đọc không còn trong bộ nhớ tạm. Vui lòng bấm Đọc file Excel lại rồi cập nhật.', version: APP_CODE_VERSION };
      if (statusUploadId) putUploadStatus_(statusUploadId, res0);
      return res0;
    }

    let parsed;
    try { parsed = JSON.parse(raw); }
    catch (e) {
      const res0 = { ok: false, msg: 'Không đọc được dữ liệu Excel đã lưu tạm. Vui lòng đọc lại file Excel.', version: APP_CODE_VERSION };
      if (statusUploadId) putUploadStatus_(statusUploadId, res0);
      return res0;
    }

    if (!parsed || parsed.ok === false) {
      const res0 = { ok: false, msg: (parsed && parsed.msg) || 'File Excel chưa được đọc thành công.', version: APP_CODE_VERSION };
      if (statusUploadId) putUploadStatus_(statusUploadId, res0);
      return res0;
    }
    const rows = Array.isArray(parsed.rows) ? parsed.rows : [];
    if (!rows.length) {
      const res0 = { ok: false, msg: 'Không có dòng hợp lệ để import. Vui lòng kiểm tra lại file Excel.', version: APP_CODE_VERSION };
      if (statusUploadId) putUploadStatus_(statusUploadId, res0);
      return res0;
    }

    const res = importDutyEntriesWithUser_(user, rows, !!p.submit);
    res.invalid = (res.invalid || []).concat(parsed.invalid || []);
    res.version = APP_CODE_VERSION;
    res.uploadId = uploadId;
    res.statusUploadId = statusUploadId;
    res.msg = res.msg || ('Đã import ' + (res.inserted || rows.length) + ' dòng từ dữ liệu Excel đã đọc.');
    if (statusUploadId) putUploadStatus_(statusUploadId, res);
    return res;
  } catch (err) {
    const res = { ok: false, msg: 'Lỗi cập nhật dữ liệu Excel: ' + err.message, version: APP_CODE_VERSION };
    if (statusUploadId) putUploadStatus_(statusUploadId, res);
    return res;
  }
}

function dutyParseUploadedSpreadsheet_(p) {
  const fileName = safeStr_(p.fileName || 'duty.xlsx');
  const mimeType = safeStr_(p.mimeType || 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet');
  let base64 = safeStr_(p.base64 || p.data || '');
  if (!base64) throw new Error('Thiếu dữ liệu file Excel.');
  const comma = base64.indexOf(',');
  if (base64.indexOf('base64,') >= 0 && comma >= 0) base64 = base64.substring(comma + 1);
  const bytes = Utilities.base64Decode(base64);
  const blob = Utilities.newBlob(bytes, mimeType, fileName);
  const ext = fileName.split('.').pop().toLowerCase();
  if (ext === 'csv' || mimeType.indexOf('csv') >= 0) {
    let text = blob.getDataAsString('UTF-8');
    text = text.replace(/^\uFEFF/, '');
    const rows = Utilities.parseCsv(text);
    return dutyTableArrayToObjects_(rows);
  }
  if (ext !== 'xlsx' && mimeType.indexOf('spreadsheet') < 0 && mimeType.indexOf('excel') < 0) {
    throw new Error('Chỉ hỗ trợ .xlsx hoặc .csv. Nếu đang dùng .xls cũ, hãy mở bằng Excel rồi Save As .xlsx.');
  }
  return dutyParseXlsxBlob_(blob);
}

function dutyParseXlsxBlob_(blob) {
  // V112: Apps Script Utilities.unzip() yêu cầu Blob có contentType = application/zip.
  // File .xlsx thực chất là một file ZIP nhưng trình duyệt thường gửi MIME là
  // application/vnd.openxmlformats-officedocument.spreadsheetml.sheet, nên V111 bị lỗi:
  // "Đối số không hợp lệ: ContentType. Phải thuộc loại: application/zip".
  const bytes = blob.getBytes();
  if (!bytes || bytes.length < 4) throw new Error('File Excel rỗng hoặc không đọc được dữ liệu.');
  // Kiểm tra chữ ký ZIP: PK hoặc PK/PK .
  if (!(bytes[0] === 80 && bytes[1] === 75)) {
    throw new Error('File không đúng định dạng .xlsx hợp lệ. Vui lòng mở file bằng Excel và Save As định dạng Excel Workbook (*.xlsx).');
  }
  const zipBlob = Utilities.newBlob(bytes, 'application/zip', blob.getName() || 'duty.xlsx');
  const parts = Utilities.unzip(zipBlob);
  const files = {};
  parts.forEach(function(b) {
    files[b.getName()] = b.getDataAsString('UTF-8');
  });
  if (!files['xl/workbook.xml'] || !files['[Content_Types].xml']) {
    throw new Error('File .xlsx không đầy đủ cấu trúc workbook. Vui lòng lưu lại file bằng Microsoft Excel/Google Sheets rồi import lại.');
  }
  const shared = dutyParseSharedStrings_(files['xl/sharedStrings.xml'] || '');
  const sheetPath = dutyFindWorksheetPath_(files, 'DUTY_IMPORT') || 'xl/worksheets/sheet1.xml';
  const sheetXml = files[sheetPath];
  if (!sheetXml) throw new Error('Không tìm thấy sheet DUTY_IMPORT hoặc sheet đầu tiên trong file Excel.');
  const values = dutyParseWorksheetXml_(sheetXml, shared);
  return dutyTableArrayToObjects_(values);
}

function dutyFindWorksheetPath_(files, wantedName) {
  try {
    const workbookXml = files['xl/workbook.xml'];
    const relsXml = files['xl/_rels/workbook.xml.rels'];
    if (!workbookXml || !relsXml) return '';
    const ns = XmlService.getNamespace('http://schemas.openxmlformats.org/spreadsheetml/2006/main');
    const rns = XmlService.getNamespace('r', 'http://schemas.openxmlformats.org/officeDocument/2006/relationships');
    const doc = XmlService.parse(workbookXml);
    const sheetsEl = doc.getRootElement().getChild('sheets', ns);
    if (!sheetsEl) return '';
    let rid = '';
    sheetsEl.getChildren('sheet', ns).some(function(sh) {
      const nm = safeStr_(sh.getAttribute('name') && sh.getAttribute('name').getValue());
      if (normalizeText_(nm) === normalizeText_(wantedName)) {
        rid = safeStr_(sh.getAttribute('id', rns) && sh.getAttribute('id', rns).getValue());
        return true;
      }
      return false;
    });
    if (!rid) return '';
    const relNs = XmlService.getNamespace('http://schemas.openxmlformats.org/package/2006/relationships');
    const relDoc = XmlService.parse(relsXml);
    const rel = relDoc.getRootElement().getChildren('Relationship', relNs).find(function(x) {
      return safeStr_(x.getAttribute('Id') && x.getAttribute('Id').getValue()) === rid;
    });
    if (!rel) return '';
    let target = safeStr_(rel.getAttribute('Target') && rel.getAttribute('Target').getValue());
    if (!target) return '';
    if (target.indexOf('/') === 0) target = target.substring(1);
    if (target.indexOf('xl/') !== 0) target = 'xl/' + target.replace(/^\.\.\//, '');
    return target;
  } catch (e) { return ''; }
}

function dutyParseSharedStrings_(xml) {
  const out = [];
  if (!safeStr_(xml)) return out;
  try {
    const ns = XmlService.getNamespace('http://schemas.openxmlformats.org/spreadsheetml/2006/main');
    const doc = XmlService.parse(xml);
    doc.getRootElement().getChildren('si', ns).forEach(function(si) {
      const t = si.getChild('t', ns);
      if (t) { out.push(t.getText()); return; }
      let s = '';
      si.getChildren('r', ns).forEach(function(r) {
        const rt = r.getChild('t', ns);
        if (rt) s += rt.getText();
      });
      out.push(s);
    });
  } catch (e) {}
  return out;
}

function dutyParseWorksheetXml_(xml, sharedStrings) {
  const ns = XmlService.getNamespace('http://schemas.openxmlformats.org/spreadsheetml/2006/main');
  const doc = XmlService.parse(xml);
  const root = doc.getRootElement();
  const sheetData = root.getChild('sheetData', ns);
  if (!sheetData) return [];
  const rows = [];
  sheetData.getChildren('row', ns).forEach(function(rowEl) {
    const row = [];
    rowEl.getChildren('c', ns).forEach(function(c) {
      const ref = safeStr_(c.getAttribute('r') && c.getAttribute('r').getValue());
      const col = dutyColumnIndexFromRef_(ref);
      row[col] = dutyCellValue_(c, ns, sharedStrings);
    });
    rows.push(row.map(function(v){ return v === undefined ? '' : v; }));
  });
  return rows;
}

function dutyCellValue_(c, ns, sharedStrings) {
  const tAttr = safeStr_(c.getAttribute('t') && c.getAttribute('t').getValue());
  if (tAttr === 'inlineStr') {
    const is = c.getChild('is', ns);
    const t = is && is.getChild('t', ns);
    return t ? t.getText() : '';
  }
  const v = c.getChild('v', ns);
  const raw = v ? v.getText() : '';
  if (tAttr === 's') return sharedStrings[Number(raw)] || '';
  if (tAttr === 'b') return raw === '1' ? 'TRUE' : 'FALSE';
  return raw;
}

function dutyColumnIndexFromRef_(ref) {
  const letters = safeStr_(ref).replace(/[^A-Z]/gi, '').toUpperCase();
  let n = 0;
  for (let i = 0; i < letters.length; i++) n = n * 26 + (letters.charCodeAt(i) - 64);
  return Math.max(0, n - 1);
}

function dutyTableArrayToObjects_(rows) {
  rows = rows || [];
  let headerRow = -1;
  for (let i = 0; i < rows.length; i++) {
    const normHeaders = (rows[i] || []).map(function(x){ return normalizeText_(x); });
    if (normHeaders.indexOf('ngay truc') >= 0 || normHeaders.indexOf('duty date') >= 0 || normHeaders.indexOf('ngay truc') >= 0) { headerRow = i; break; }
    if (normHeaders.indexOf('ho ten') >= 0 && (normHeaders.indexOf('don vi') >= 0 || normHeaders.indexOf('unit code') >= 0)) { headerRow = i; break; }
  }
  if (headerRow < 0) headerRow = 0;
  const headers = (rows[headerRow] || []).map(function(h){ return safeStr_(h); });
  const out = [];
  for (let r = headerRow + 1; r < rows.length; r++) {
    const row = rows[r] || [];
    if (row.every(function(v){ return safeStr_(v) === ''; })) continue;
    const obj = {};
    headers.forEach(function(h, i){ if (h) obj[h] = row[i] === undefined ? '' : row[i]; });
    out.push(obj);
  }
  return out;
}

function dutyNormalizeServerImportRows_(rawRows, user, checkPermission) {
  const valid = [];
  const invalid = [];
  (rawRows || []).forEach(function(row, idx) {
    const dutyDate = dutyExcelDateToIsoServer_(dutyGetImportCell_(row, ['NGAY_TRUC','Ngày trực','Ngay truc','DUTY_DATE','date']));
    const unitCode = normalizeDutyUnitCode_(dutyGetImportCell_(row, ['UNIT_CODE','Mã đơn vị','Ma don vi']), dutyGetImportCell_(row, ['DON_VI','Đơn vị','Don vi','Trụ sở','Tru so','UNIT_NAME']));
    const unitLabel = dutyUnitName_(unitCode) || safeStr_(dutyGetImportCell_(row, ['DON_VI','Đơn vị','Don vi','Trụ sở','Tru so','UNIT_NAME']));
    const fullname = safeStr_(dutyGetImportCell_(row, ['HO_TEN','Họ tên','Ho ten','FULLNAME','Tên','Ten']));
    const position = safeStr_(dutyGetImportCell_(row, ['CHUC_VU','Chức vụ','Chuc vu','POSITION']));
    const phone = safeStr_(dutyGetImportCell_(row, ['SO_DIEN_THOAI','Số điện thoại','So dien thoai','PHONE','SĐT','SDT']));
    const note = safeStr_(dutyGetImportCell_(row, ['GHI_CHU','Ghi chú','Ghi chu','NOTE']));
    const dutyType = safeStr_(dutyGetImportCell_(row, ['DUTY_TYPE','Loại ngày','Loai ngay'])) || 'Thứ 7/CN';
    const dutyShift = safeStr_(dutyGetImportCell_(row, ['DUTY_SHIFT','Ca trực','Ca truc'])) || 'Cả ngày';
    const dutyRole = safeStr_(dutyGetImportCell_(row, ['DUTY_ROLE','Vai trò','Vai tro']));
    if (!dutyDate || !unitCode || !fullname) {
      invalid.push({ row: idx + 2, msg: 'Thiếu NGAY_TRUC/UNIT_CODE hoặc DON_VI/HO_TEN' });
      return;
    }
    if (checkPermission && !canEditDutyUnit_(user, unitCode, unitLabel)) {
      invalid.push({ row: idx + 2, msg: 'Không có quyền cập nhật đơn vị ' + unitLabel });
      return;
    }
    valid.push({ dutyDate: dutyDate, unitCode: unitCode, unitName: unitLabel, fullname: fullname, position: position, phone: phone, note: note, dutyType: dutyType, dutyShift: dutyShift, dutyRole: dutyRole });
  });
  return { rows: valid, invalid: invalid };
}

function dutyGetImportCell_(row, names) {
  row = row || {};
  const keys = Object.keys(row);
  for (let i = 0; i < names.length; i++) {
    const want = normalizeText_(names[i]);
    for (let k = 0; k < keys.length; k++) {
      if (normalizeText_(keys[k]) === want) return row[keys[k]];
    }
  }
  return '';
}

function dutyExcelDateToIsoServer_(v) {
  if (v instanceof Date && !isNaN(v.getTime())) return Utilities.formatDate(v, CFG.TZ, 'yyyy-MM-dd');
  const s = safeStr_(v);
  if (!s) return '';
  if (/^\d+(\.\d+)?$/.test(s)) {
    const n = Number(s);
    if (n > 20000 && n < 80000) {
      const ms = Math.round((n - 25569) * 86400000);
      return Utilities.formatDate(new Date(ms), 'GMT', 'yyyy-MM-dd');
    }
  }
  return isoDate_(s) || s;
}
